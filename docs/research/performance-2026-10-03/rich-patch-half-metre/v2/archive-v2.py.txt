def add(original,target,scope,expected=None,compress=False):
 o=Path(original); data=o.read_bytes()
 if expected: assert sha(data)==expected,(str(o),sha(data),expected)
 dest=OUT/target; assert not dest.exists(),target
 dest.parent.mkdir(parents=True,exist_ok=True)

 if compress or o.suffix=='.gz':
  raw=gzip.decompress(stored)
  if compress: assert raw==data
  r.update(decompressedBytes=len(raw),decompressedSha256=sha(raw))
 records.append(r); return r
