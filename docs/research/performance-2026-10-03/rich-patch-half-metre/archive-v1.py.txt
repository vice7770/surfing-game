sha=lambda b:hashlib.sha256(b).hexdigest()
records=[]
def add(original,target,scope,expected=None,compress=False):
    original=Path(original); data=original.read_bytes()
    if expected: assert sha(data)==expected,(str(original),sha(data),expected)
    dest=OUT/target; assert not dest.exists(),target
    dest.parent.mkdir(parents=True,exist_ok=True)
    stored=gzip.compress(data,mtime=0) if compress else data
    dest.write_bytes(stored)
    r={'path':target,'originalPath':str(original),'scope':scope,'originalBytes':len(data),'originalSha256':sha(data),'storedBytes':len(stored),'storedSha256':sha(stored),'encoding':'deterministic-gzip' if compress else ('original-gzip' if original.suffix=='.gz' else 'original-bytes')}
    if compress or original.suffix=='.gz':
        raw=gzip.decompress(stored)
        if compress: assert raw==data
        r.update(decompressedBytes=len(raw),decompressedSha256=sha(raw))
    assert dest.read_bytes()==stored
    records.append(r)
    return r
