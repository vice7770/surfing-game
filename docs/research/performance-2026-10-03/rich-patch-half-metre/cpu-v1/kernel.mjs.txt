export const SPEC = Object.freeze({ schema: 1, size: 96, baselineSpacing: .25, candidateSpacing: .5, sampleSpacing: .125, deadlineMs: 10000,
  limits: ['CPU/F64 height-only approximation; not GPU/raster parity.', 'No mask, depth, foam, carved-tube/cavity quality or FPS acceptance.', 'One held state; no solver, browser or water advance.'] });
export const sha256 = bytes => createHash('sha256').update(bytes).digest('hex');
const check = (condition, reason) => { if (!condition) throw Error(reason); };

// The exact baseline/candidate factory arrays are supplied as reviewed metadata by the
// actual held-render factory. This helper imports no second Three class or generator.
export function decodeTyped(field, expectedType) {
  check(field && field.type === expectedType && typeof field.base64 === 'string', 'Missing exact '+expectedType+' array');
  const bytes = Buffer.from(field.base64, 'base64');
  const type = { Float32Array, Uint16Array, Uint32Array }[expectedType];
  check(type && field.byteLength === bytes.length && field.length * type.BYTES_PER_ELEMENT === bytes.length, 'Typed byte length mismatch');
  check(/^[0-9a-f]{64}$/.test(field.sha256 ?? '') && sha256(bytes) === field.sha256, 'Typed bytes hash mismatch');
  const copy = Uint8Array.from(bytes);
  return new type(copy.buffer);
}

export function validateTopology(geometry) {
  const { size, spacing, position, skirt, index } = geometry;
  const n = Math.round(size / spacing) + 1, top = n*n, rimLength = 4*(n-1);
  check(Number.isFinite(size) && size > 0 && Number.isFinite(spacing) && spacing > 0 && n >= 2 && size === (n-1)*spacing, 'Invalid exact patch lattice');
  check(position instanceof Float32Array && skirt instanceof Float32Array && (index instanceof Uint16Array || index instanceof Uint32Array), 'Factory array types differ');
  check(position.length === 3*(top+rimLength) && skirt.length === top+rimLength && index.length === 6*(n-1)*(n-1)+6*rimLength, 'Factory layout/count mismatch');
  for(let j=0;j<n;j++)for(let i=0;i<n;i++){
    const k=j*n+i;
    check(position[3*k] === Math.fround(-size/2+i*spacing) && position[3*k+1] === 0 && position[3*k+2] === Math.fround(-size/2+j*spacing) && skirt[k] === 0, 'Top position/skirt layout mismatch');
  }
  let o=0;
  const same = (...values) => { for(const value of values)check(index[o++] === value, 'Factory triangle diagonal/winding mismatch'); };
  for(let j=0;j<n-1;j++)for(let i=0;i<n-1;i++){
    const a=j*n+i,b=a+1,c=a+n,d=c+1;
    same(a,c,b,b,c,d);
  }
  const rim=[];
  for(let i=0;i<n-1;i++)rim.push(i);
  for(let j=0;j<n-1;j++)rim.push(j*n+n-1);
  for(let i=n-1;i>0;i--)rim.push((n-1)*n+i);
  for(let j=n-1;j>0;j--)rim.push(j*n);
  for(let k=0;k<rim.length;k++){
    const a=rim[k],b=top+k;
    check(skirt[b] === 1 && position[3*b] === position[3*a] && position[3*b+1] === 0 && position[3*b+2] === position[3*a+2], 'Factory rim/skirt mismatch');
    same(a,rim[(k+1)%rim.length],b,rim[(k+1)%rim.length],top+(k+1)%rim.length,b);
  }
  check(o===index.length, 'Unvalidated trailing indices');
  return { n, topVertexCount: top, rimVertexCount: rimLength, vertexCount: skirt.length, triangleCount: index.length/3 };
}

export function triangleHeight(a,b,c,d,tx,tz) {
  check(tx>=0 && tx<=1 && tz>=0 && tz<=1, 'Triangle coordinates outside cell');
  // a,c,b followed by b,c,d: the diagonal is b-c, not a-d.
  return tx+tz<=1 ? a+tx*(b-a)+tz*(c-a) : d+(1-tx)*(c-d)+(1-tz)*(b-d);
}

export function planarHeight(mesh,x,z) {
  const { n, rect, heights }=mesh, gx=(x-rect[0])/(rect[2]-rect[0])*(n-1),gz=(z-rect[1])/(rect[3]-rect[1])*(n-1);
  check(gx>=-1e-9 && gx<=n-1+1e-9 && gz>=-1e-9 && gz<=n-1+1e-9, 'Point outside actual patch');
  const ix=Math.min(n-2,Math.max(0,Math.floor(gx))),iz=Math.min(n-2,Math.max(0,Math.floor(gz))),tx=Math.min(1,Math.max(0,gx-ix)),tz=Math.min(1,Math.max(0,gz-iz));
  const a=iz*n+ix,b=a+1,c=a+n,d=c+1;
  return {height:triangleHeight(heights[a],heights[b],heights[c],heights[d],tx,tz), triangle:2*(iz*(n-1)+ix)+(tx+tz<=1?0:1), vertexIndices:tx+tz<=1?[a,c,b]:[b,c,d]};
}

function weights(t) { const t2=t*t,t3=t2*t;return[(-t3+2*t2-t)/2,(3*t3-5*t2+2)/2,(-3*t3+4*t2+t)/2,(t3-t2)/2]; }
export function cubicHeight(data,grid,x,z) {
  const gx=(x-grid.xMin)/grid.spacing,gz=(z-grid.zMin)/grid.spacing,ix=Math.floor(gx),iz=Math.floor(gz),wx=weights(gx-ix),wz=weights(gz-iz);
  let height=0;
  for(let j=0;j<4;j++)for(let i=0;i<4;i++){
    const row=Math.min(grid.nz-1,Math.max(0,iz+j-1)),column=Math.min(grid.nx-1,Math.max(0,ix+i-1));
    height+=wz[j]*wx[i]*data[2*(row*grid.nx+column)];
  }
  return height;
}

function underlyingNodes(data,grid,x,z) {
  const ix=Math.floor((x-grid.xMin)/grid.spacing),iz=Math.floor((z-grid.zMin)/grid.spacing),nodes=[];
  for(let j=0;j<4;j++)for(let i=0;i<4;i++){
    const row=Math.min(grid.nz-1,Math.max(0,iz+j-1)),column=Math.min(grid.nx-1,Math.max(0,ix+i-1));
    nodes.push({column,row,height:data[2*(row*grid.nx+column)]});
  }return nodes;
}

// Fail closed on unsupported actual transforms. The canonical patch translates/scales
// XZ and does not rotate into Y; retain the shader's local-height→world-Y operation.
export function validateTransform(matrix,rect,size) {
  check(Array.isArray(matrix)&&matrix.length===16&&matrix.every(Number.isFinite), 'Missing actual patch matrixWorld');
  check([1,2,3,4,6,7,8,9,11].every(k=>matrix[k]===0)&&matrix[15]===1&&matrix[0]>0&&matrix[5]>0&&matrix[10]>0, 'Unsupported actual patch transform');
  const expected=[matrix[12]-matrix[0]*size/2,matrix[14]-matrix[10]*size/2,matrix[12]+matrix[0]*size/2,matrix[14]+matrix[10]*size/2];
  check(Array.isArray(rect)&&rect.length===4&&rect.every((v,i)=>Number.isFinite(v)&&Math.abs(v-expected[i])<=1e-9), 'Actual patch rect disagrees with transform');
  return expected;
}

function lift(geometry,matrix,rect,data,grid,poll) {
  const topology=validateTopology(geometry),heights=new Float64Array(topology.topVertexCount);let lo=Infinity,hi=-Infinity;
  for(let j=0;j<topology.n;j++){
    poll();
    for(let i=0;i<topology.n;i++){
      const k=j*topology.n+i,x=matrix[0]*geometry.position[3*k]+matrix[12],z=matrix[10]*geometry.position[3*k+2]+matrix[14],h=matrix[5]*cubicHeight(data,grid,x,z)+matrix[13];
      check(Number.isFinite(h),'Nonfinite lifted height');heights[k]=h;lo=Math.min(lo,h);hi=Math.max(hi,h);
    }
  }
  return{...topology,rect,heights,heightRange:[lo,hi],arrayHashes:{position:sha256(new Uint8Array(geometry.position.buffer,geometry.position.byteOffset,geometry.position.byteLength)),skirt:sha256(new Uint8Array(geometry.skirt.buffer,geometry.skirt.byteOffset,geometry.skirt.byteLength)),index:sha256(new Uint8Array(geometry.index.buffer,geometry.index.byteOffset,geometry.index.byteLength))}};
}

function distribution(capacity) {
  const absolute=new Float64Array(capacity);let count=0,sum=0,square=0,max=null;
  return{add(value,at){absolute[count++]=Math.abs(value);sum+=value;square+=value*value;if(!max||Math.abs(value)>max.absolute)max={value,absolute:Math.abs(value),...at};},finish(){const sorted=absolute.subarray(0,count).sort(),q=p=>sorted[Math.ceil(p*(count-1))];return{count,signedMean:sum/count,rms:Math.sqrt(square/count),p50Absolute:q(.5),p95Absolute:q(.95),p99Absolute:q(.99),max};}};
}

export function analyzeHeightRisk(input,{now=()=>performance.now(),deadlineMs=SPEC.deadlineMs}={}) {
  const begun=now(),poll=()=>check(now()-begun<=deadlineMs,'Hard CPU deadline exceeded');
  check(deadlineMs===SPEC.deadlineMs,'Fixed10-second calculation bound required');
  const {observed,grid,data,baseline,candidate,matrix,rect,authority}=input;
  check(observed?.waterTubeCount===0 && observed.waterPatchActive===1 && observed.waterLook==='rich' && observed.vertexNormals===false, 'Actual zero-tube Rich/pixel-normal witness absent');
  check(observed.uniformWitness==='actual-material-uniforms-after-real-draw','Expected counts are not actual material uniform authority');
  check(data instanceof Float32Array && Number.isInteger(grid?.nx)&&Number.isInteger(grid?.nz)&&grid.nx>=2&&grid.nz>=2&&grid.spacing>0&&Number.isFinite(grid.xMin)&&Number.isFinite(grid.zMin)&&data.length===2*grid.nx*grid.nz,'Surface texture/grid mismatch');
  check(baseline.size===SPEC.size && candidate.size===SPEC.size && baseline.spacing===SPEC.baselineSpacing && candidate.spacing===SPEC.candidateSpacing,'Only predeclared96m/.25→.5 comparison allowed');
  validateTransform(matrix,rect,SPEC.size);
  const fine=lift(baseline,matrix,rect,data,grid,poll),coarse=lift(candidate,matrix,rect,data,grid,poll);
  const original=distribution(fine.topVertexCount);
  for(let j=0;j<fine.n;j++){
    poll();
    for(let i=0;i<fine.n;i++){
      const k=j*fine.n+i,x=matrix[0]*baseline.position[3*k]+matrix[12],z=matrix[10]*baseline.position[3*k+2]+matrix[14],p=planarHeight(coarse,x,z);
      original.add(p.height-fine.heights[k],{x,z,baselineVertex:k,candidateTriangle:p.triangle});
    }
  }
  const nx=Math.round((rect[2]-rect[0])/SPEC.sampleSpacing)+1,nz=Math.round((rect[3]-rect[1])/SPEC.sampleSpacing)+1;
  check(nx*nz<=600000 && Math.abs((nx-1)*SPEC.sampleSpacing-(rect[2]-rect[0]))<1e-9 && Math.abs((nz-1)*SPEC.sampleSpacing-(rect[3]-rect[1]))<1e-9,'Fixed.125m lattice exceeds exact declared domain/bound');
  const delta=distribution(nx*nz),fineError=distribution(nx*nz),coarseError=distribution(nx*nz);
  for(let j=0;j<nz;j++){
    poll();
    for(let i=0;i<nx;i++){
      const x=rect[0]+i*SPEC.sampleSpacing,z=rect[1]+j*SPEC.sampleSpacing,a=planarHeight(fine,x,z),b=planarHeight(coarse,x,z),h=matrix[5]*cubicHeight(data,grid,x,z)+matrix[13],at={x,z,baselineTriangle:a.triangle,candidateTriangle:b.triangle};
      delta.add(b.height-a.height,at);fineError.add(a.height-h,at);coarseError.add(b.height-h,at);
    }
  }
  poll();const metrics={atOriginalQuarterMetreNodes:original.finish(),fixedEighthMetreLattice:{nx,nz,spacing:SPEC.sampleSpacing,delta:delta.finish(),baselineErrorToCubic:fineError.finish(),candidateErrorToCubic:coarseError.finish()}};poll();
  for(const m of[metrics.atOriginalQuarterMetreNodes,metrics.fixedEighthMetreLattice.delta,metrics.fixedEighthMetreLattice.baselineErrorToCubic,metrics.fixedEighthMetreLattice.candidateErrorToCubic])m.max.underlyingTextureNodes=underlyingNodes(data,grid,m.max.x,m.max.z);
  const descriptor=mesh=>({n:mesh.n,topVertexCount:mesh.topVertexCount,rimVertexCount:mesh.rimVertexCount,vertexCount:mesh.vertexCount,triangleCount:mesh.triangleCount,heightRange:mesh.heightRange,arrayHashes:mesh.arrayHashes});
  return{schema:SPEC.schema,status:'CPU-only height supplement completed',authority,observed,rect,matrix,grid,spec:SPEC,geometry:{baseline:descriptor(fine),candidate:descriptor(coarse)},metrics,
    skirt:{excludedFromTopSurfaceStatistics:true,localDropMeters:.3,worldDropMeters:matrix[5]*.3,note:'Same endpoint displacement/drop rule; skirt raster coverage not tested.'},elapsedCalculationMs:now()-begun};
}

function loadHashGuarded(path,expected) {
  check(path && /^[0-9a-f]{64}$/.test(expected??''),'Explicit reviewed path+SHA256 required');
  const raw=readFileSync(path);check(sha256(raw)===expected,'Reviewed source/factory file hash changed');
  const bytes=path.endsWith('.gz')?gunzipSync(raw):raw;
  return{json:JSON.parse(bytes),compressedSha256:expected,contentSha256:sha256(bytes),rawBytes:raw.length,jsonBytes:bytes.length};
}

// The fresh capture adapter contract is intentionally explicit: no fallback to older
// archives, undefined uniforms, capacities, closest objects, or guessed factory data.
export function normalizeFreshCapture(source,authority,expectedFactorySha256) {
  const factory=source.patchFactory,patch=source.observed?.patch;
  check(factory && patch && source.scalars?.waterUniforms,'Fresh actual uniforms/patch/factory metadata required');
  check(typeof factory.factorySource==='string' && /^[0-9a-f]{64}$/.test(expectedFactorySha256??'') && sha256(Buffer.from(factory.factorySource))===expectedFactorySha256 && factory.factorySourceSha256===expectedFactorySha256,'Reviewed embedded factory source hash differs');
  check(factory.bindsLoadedConstructors===true && factory.nativeByteMatch===true,'Factory must bind loaded constructors and byte-match actual native geometry');
  const object=source.scalars?.scene?.objects?.find(o=>o.id===patch.objectId);
  check(object && object.geometry===patch.geometryId && object.visible,'Fresh actual visible patch object/geometry missing');
  check(JSON.stringify(object.matrixWorld)===JSON.stringify(patch.matrixWorld),'Fresh actual patch transform witnesses disagree');
  const originalMeta=source.scalars.scene.geometries.find(g=>g.id===patch.geometryId);
  check(originalMeta && originalMeta.drawRange.start===0,'Actual patch draw range invalid');
  const quarter=factory.geometries?.quarter,half=factory.geometries?.half;
  check(quarter && half,'Quarter/half actual factory arrays missing');
  const original=source.arrays?.['scene.geometry.'+patch.geometryId+'.position'];
  check(original && original.sha256===quarter.attributes.position.sha256,'Factory baseline does not match actual captured patch positions');
  for(const name of ['index','skirt','onPatch'])check(source.arrays['scene.geometry.'+patch.geometryId+'.'+name]?.sha256===(name==='index'?quarter.index:quarter.attributes[name])?.sha256,'Factory baseline '+name+' differs from actual captured patch');
  const value=field=>field && typeof field==='object' && Object.hasOwn(field,'value') ? field.value : field;
  const observed={waterTubeCount:value(source.scalars.waterUniforms.waterTubeCount),waterPatchActive:value(source.scalars.waterUniforms.waterPatchActive),waterPatchRect:value(source.scalars.waterUniforms.waterPatchRect),waterLook:source.observed.waterLook,vertexNormals:source.observed.vertexNormals,uniformWitness:'actual-material-uniforms-after-real-draw'};
  const decodeGeometry=o=>{
    const position=decodeTyped(o.attributes.position,'Float32Array'),skirt=decodeTyped(o.attributes.skirt,'Float32Array'),onPatch=decodeTyped(o.attributes.onPatch,'Float32Array');
    check(onPatch.length===skirt.length && onPatch.every(v=>v===1),'Factory onPatch flag differs from actual Rich patch');
    return{size:o.size,spacing:o.spacing,position,skirt,index:decodeTyped(o.index,o.index.type)};
  };
  return{authority:{...authority,factorySourceSha256:expectedFactorySha256,sourceHashes:factory.sourceHashes,seaTime:source.scalars.status.seaTime},observed,
    grid:source.scalars.waterGrid,data:decodeTyped(source.arrays['water.surfaceData'],'Float32Array'),matrix:object.matrixWorld,rect:observed.waterPatchRect,
    baseline:decodeGeometry(quarter),candidate:decodeGeometry(half)};
}

async function main(){
  const args=Object.fromEntries(process.argv.slice(2).map(arg=>{check(arg.startsWith('--')&&arg.includes('='),'Use --name=value');const i=arg.indexOf('=');return[arg.slice(2,i),arg.slice(i+1)];}));
  const plan={spec:SPEC,status:'prepared only; no scene/source loaded',sourceSchema:'Fresh lossless held scalars/arrays; scalars.waterUniforms actual values; observed.patch:{objectId,geometryId,matrixWorld}; observed.waterLook/vertexNormals.',factorySchema:'Embedded patchFactory:{factorySource,factorySourceSha256,bindsLoadedConstructors:true,nativeByteMatch:true,sourceHashes,geometries:{quarter,half}}; each geometry:{size,spacing,attributes:{position,skirt,onPatch exact typed fields},index exact typed field}',helperSha256:sha256(readFileSync(fileURLToPath(import.meta.url))),command:'node /private/tmp/rich-patch-height-risk.mjs --run=true --source=<fresh.json.gz> --sourceSha256=<reviewed> --factorySha256=<reviewed embedded source> --out=<fresh report.json>'};
  if(args.run!=='true'){console.log(JSON.stringify(plan,null,2));return;}
  check(args.out && !existsSync(args.out),'Fresh report output required');
  const report={...plan,status:'incomplete',pass:false};
  try{const source=loadHashGuarded(args.source,args.sourceSha256);const authority={source:{path:resolve(args.source),...Object.fromEntries(Object.entries(source).filter(([k])=>k!=='json'))}};report.result=analyzeHeightRisk(normalizeFreshCapture(source.json,authority,args.factorySha256));report.status=report.result.status;report.pass=true;}
  catch(error){report.error=String(error);process.exitCode=1;}
  finally{writeFileSync(args.out,JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify({pass:report.pass,status:report.status,error:report.error,out:args.out}));}
}
if(process.argv[1]&&resolve(process.argv[1])===fileURLToPath(import.meta.url))await main();
