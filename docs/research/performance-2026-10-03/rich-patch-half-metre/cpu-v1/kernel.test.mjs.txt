import test from 'node:test';
import assert from 'node:assert/strict';
import {triangleHeight,planarHeight,validateTopology,validateTransform,cubicHeight,normalizeFreshCapture} from './rich-patch-height-risk.mjs';

// Independent explicit one-cell topology, including the four rim faces.
const tiny=()=>({size:1,spacing:1,
 position:new Float32Array([-.5,0,-.5,.5,0,-.5,-.5,0,.5,.5,0,.5,-.5,0,-.5,.5,0,-.5,.5,0,.5,-.5,0,.5]),
 skirt:new Float32Array([0,0,0,0,1,1,1,1]),
 index:new Uint16Array([0,2,1,1,2,3,0,1,4,1,5,4,1,3,5,3,6,5,3,2,6,2,7,6,2,0,7,0,4,7])});

test('actual b-c diagonal creates two planes, distinct from bilinear and a-d facets',()=>{
  assert.equal(triangleHeight(0,0,0,1,.25,.25),0); // Lower triangle, all three corners zero.
  assert.equal(triangleHeight(0,0,0,1,.75,.75),.5); // Upper triangle: d has barycentric weight .5.
  assert.equal(triangleHeight(0,0,0,1,.5,.5),0); // Both b-c endpoints are zero.
  assert.notEqual(triangleHeight(0,0,0,1,.25,.25),.25*.25);
  assert.notEqual(triangleHeight(0,0,0,1,.5,.5),.5); // Opposite a-d diagonal would pass through height .5.
});

test('affine plane reproduced on either triangle with original corner nodes',()=>{
  // h=4+2*x-3*z, corners a4,b6,c1,d3.
  assert.equal(triangleHeight(4,6,1,3,.25,.25),3.75);
  assert.equal(triangleHeight(4,6,1,3,.75,.75),3.25);
  for(const [x,z,expected]of[[0,0,4],[1,0,6],[0,1,1],[1,1,3]])assert.equal(triangleHeight(4,6,1,3,x,z),expected);
});

test('original quarter-node positions select the correct facet and preserve patch endpoints',()=>{
  const mesh={n:2,rect:[10,-4,11,-3],heights:new Float64Array([0,0,0,1])};
  assert.deepEqual(planarHeight(mesh,10.25,-3.75),{height:0,triangle:0,vertexIndices:[0,2,1]});
  assert.deepEqual(planarHeight(mesh,10.75,-3.25),{height:.5,triangle:1,vertexIndices:[1,2,3]});
  assert.equal(planarHeight(mesh,11,-3).height,1);
  assert.throws(()=>planarHeight(mesh,9,-4),/outside actual patch/);
});

test('canonical one-cell top/rim indices accepted; a-d diagonal and broken skirt rejected',()=>{
  assert.deepEqual(validateTopology(tiny()),{n:2,topVertexCount:4,rimVertexCount:4,vertexCount:8,triangleCount:10});
  const wrong=tiny();wrong.index.set([0,3,1,0,2,3]);assert.throws(()=>validateTopology(wrong),/diagonal\/winding/);
  const badSkirt=tiny();badSkirt.position[12]=.4;assert.throws(()=>validateTopology(badSkirt),/rim\/skirt/);
});

test('actual scaled/transformed rectangle is required; rotated/mismatched rectangle rejected',()=>{
  const matrix=[2,0,0,0,0,1,0,0,0,0,3,0,10,0,-4,1];
  assert.deepEqual(validateTransform(matrix,[9,-5.5,11,-2.5],1),[9,-5.5,11,-2.5]);
  assert.throws(()=>validateTransform(matrix,[0,0,1,1],1),/disagrees/);
  const rotated=matrix.slice();rotated[2]=1;assert.throws(()=>validateTransform(rotated,[9,-5.5,11,-2.5],1),/Unsupported/);
});

test('F64 cubic height reproduces independent constant and affine interior texture fields',()=>{
  const grid={xMin:0,zMin:0,spacing:1,nx:4,nz:4},constant=new Float32Array(32),plane=new Float32Array(32);
  for(let z=0;z<4;z++)for(let x=0;x<4;x++){constant[2*(4*z+x)]=2;plane[2*(4*z+x)]=4+2*x-3*z;}
  assert.equal(cubicHeight(constant,grid,1.25,1.25),2);
  assert.equal(cubicHeight(plane,grid,1.25,1.25),2.75);
});

test('old archive without observed fresh material uniforms/factory schema fails closed',()=>{
  assert.throws(()=>normalizeFreshCapture({scalars:{},arrays:{}},{},''),/Fresh actual uniforms\/patch\/factory/);
});
