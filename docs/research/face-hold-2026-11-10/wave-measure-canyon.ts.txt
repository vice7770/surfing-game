/**
 * face-hold calibration, the Canyon's version B (2026-11-10): its wave where it is ridden, from the solver only (no rider,
 * no runner). The sea is built as the game builds the Canyon with `?canyonWaves=true` (`canyonBed: 'b'`): Practice (the
 * default: PRACTICE_SWELL, Hs 1.4 m at the tank's edge, Tp 12 s, s 150, band 0.08, square) or a Surf-screen size (SWELLS:
 * Big is Hs 2.4 m deep-water, Tp 14 s, s 150), mid tide, calm, stage 2, stepped on the CPU at 1/30 s, and read through the
 * water the bodies read (PhysicalSurfWater.forSimulation), every FRAME s along z in columns from the rider's start, past
 * the take-off (x -112, the upcoast peak's tip), down its +x arm into the bowl. Per frame and column: every crest in the
 * window, the trough ahead, the face between (length, steepest slope and where, the fall line there), the still depth, the
 * water's flow at crest, steepest point and trough, along the fall line and along the crest.
 *   (copy to scripts/zz-wave-measure-canyon.ts)
 *   npx rolldown scripts/zz-wave-measure-canyon.ts -o out.mjs --format esm --platform node \
 *     && node out.mjs --swell practice|small|medium|big --seed 1 [--settle 40] [--seconds 180] --json out.json
 */
import { writeFileSync } from 'node:fs';
import { PhysicalSurfWater } from '../src/physics/PhysicalSurfWater';
import { createWaterSample } from '../src/physics/SurfWater';
import { SurfZoneSimulation, takeOffPoint, rideLineupOffset, rideLineupAlong, type SurfZoneConfig } from '../src/wave/SurfZoneSimulation';

const option = (name: string) => { const i = process.argv.indexOf(`--${name}`); return i >= 0 ? process.argv[i + 1] : undefined; };
const swell = option('swell') ?? 'practice';
const seed = Number(option('seed') ?? 1);
const settle = Number(option('settle') ?? 40);
const seconds = Number(option('seconds') ?? 180);
const FRAME = 0.2; const DZ = 0.25; const SEAWARD = Number(option('seaward') ?? 90); const SHOREWARD = Number(option('shoreward') ?? 45);
const DEG = 180 / Math.PI;
const SEAS: Record<string, Partial<SurfZoneConfig>> = {
  practice: { significantHeight: 1.4, peakPeriod: 12, spreading: 150, bandwidth: 0.08, heightAt: 'edge' },
  small: { significantHeight: 0.9, peakPeriod: 9, spreading: 150 },
  medium: { significantHeight: 1.4, peakPeriod: 11, spreading: 150 },
  big: { significantHeight: 2.4, peakPeriod: 14, spreading: 150 },
};
const config = { spot: 'canyon', seed, directionDegrees: 0, tide: 0, windSpeed: 0, stage: 2, compute: 'cpu', canyonBed: 'b', ...SEAS[swell] } as unknown as SurfZoneConfig;
const takeOff = takeOffPoint(config);
const start = { x: takeOff.x + rideLineupAlong(config), z: takeOff.z - rideLineupOffset(config) };
const COLUMNS = (option('columns') ?? '-10,0,10,20,30,40,50,60,70,80').split(',').map((d) => takeOff.x + Number(d));

const t0 = Date.now();
const sim = new SurfZoneSimulation(config);
const { solver } = sim;
const water = PhysicalSurfWater.forSimulation(sim);
const strength = sim.breaking.strength;
console.error(`built ${swell} seed ${seed}: nx ${solver.nx} nz ${solver.nz} dx ${solver.dx} t ${solver.time.toFixed(1)} s, ${((Date.now() - t0) / 1000).toFixed(1)} s wall; take-off ${takeOff.x}, ${takeOff.z}; start ${start.x}, ${start.z}`);

const sample = createWaterSample();
interface Point { z: number; eta: number; sx: number; sz: number; u: number; w: number; still: number; brk: number }
const read = (x: number, z: number): Point => {
  water.sampleAt(x, 50, z, sample);
  return { z, eta: sample.surfaceY, sx: sample.slopeX, sz: sample.slopeZ, u: sample.flowX, w: sample.flowZ, still: sample.stillDepth, brk: strength[solver.cellIndex(x, z)] ?? 0 };
};
const frames: Record<string, unknown>[] = [];
const step = 1 / 30;
const every = Math.round(FRAME / step);
const total = Math.round((settle + seconds) / step);
const r = (v: number, d = 2) => Math.round(v * 10 ** d) / 10 ** d;
for (let n = 0; n < total; n += 1) {
  sim.step(step);
  if (solver.time < settle || n % every) continue;
  for (const x of COLUMNS) {
    const points: Point[] = [];
    for (let z = takeOff.z - SEAWARD; z <= takeOff.z + SHOREWARD; z += DZ) points.push(read(x, z));
    for (let i = 2; i < points.length - 2; i += 1) {
      const p = points[i];
      if (!(p.eta >= points[i - 1].eta && p.eta > points[i + 1].eta)) continue;
      let trough = i;
      for (let k = i + 1; k < points.length; k += 1) {
        if (points[k].eta < points[trough].eta) trough = k;
        else if (points[k].eta > points[trough].eta + 0.05) break;
      }
      const height = p.eta - points[trough].eta;
      if (height < 0.4) continue;
      let steep = i; let grade = 0; let brk = 0;
      for (let k = i; k <= trough; k += 1) {
        const g = Math.hypot(points[k].sx, points[k].sz);
        if (g > grade) { grade = g; steep = k; }
        brk = Math.max(brk, points[k].brk);
      }
      const s = points[steep];
      // Downhill at the steepest point: the face's fall line. `fall`: its angle from +z toward +x, which is also the crest
      // line's angle to the x axis there. Along the crest: the unit vector a quarter turn from it, toward +x.
      const nxDir = -s.sx / grade; const nzDir = -s.sz / grade;
      const down = (q: Point) => q.u * nxDir + q.w * nzDir;
      const along = (q: Point) => q.u * nzDir - q.w * nxDir;
      frames.push({
        t: r(solver.time), x, z: r(p.z), etaC: r(p.eta), etaT: r(points[trough].eta), H: r(height), Lz: r(points[trough].z - p.z), slope: r(Math.atan(grade) * DEG, 1),
        fall: r(Math.atan2(nxDir, nzDir) * DEG, 1), steepFrac: r((s.eta - points[trough].eta) / height), zs: r(s.z),
        dCrest: r(p.still), dSteep: r(s.still), dTrough: r(points[trough].still),
        uCrest: r(down(p)), uSteep: r(down(s)), uTrough: r(down(points[trough])), aCrest: r(along(p)), aSteep: r(along(s)), aTrough: r(along(points[trough])),
        brkCrest: r(p.brk), brk: r(brk),
      });
    }
  }
}
console.error(`ran to ${solver.time.toFixed(1)} s in ${((Date.now() - t0) / 1000).toFixed(1)} s wall`);
writeFileSync(option('json') ?? '/dev/stdout', JSON.stringify({ swell, seed, config, takeOff, start, columns: COLUMNS, frames }));
