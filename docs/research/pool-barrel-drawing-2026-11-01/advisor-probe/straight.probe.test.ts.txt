function dist(ax: number, ay: number, bx: number, by: number) { return Math.hypot(ax - bx, ay - by); }
function segDist(px: number, py: number, ax: number, ay: number, bx: number, by: number) {
  const dx = bx - ax, dy = by - ay, l = dx * dx + dy * dy; let t = l > 0 ? ((px - ax) * dx + (py - ay) * dy) / l : 0; t = Math.max(0, Math.min(1, t));
  return dist(px, py, ax + t * dx, ay + t * dy);
}
function arc(p: ArrayLike<number>, a: number, b: number) { let s = 0; for (let i = a; i < b; i++) s += dist(p[2*i], p[2*i+1], p[2*i+2], p[2*i+3]); return s; }
function measure(p: ArrayLike<number>) {
  const x = (i: number) => p[2 * i], y = (i: number) => p[2 * i + 1];

  let minY = Infinity; for (let i = LANDMARK.crest; i < PROFILE_POINTS; i++) minY = Math.min(minY, y(i));

  // void polygon 64..112 closed: area and second moments
  let A = 0, cx = 0, cy = 0;
  const n0 = 64, n1 = 112;
  const idx: number[] = []; for (let i = n0; i <= n1; i++) idx.push(i);
  for (let k = 0; k < idx.length; k++) { const i = idx[k], j = idx[(k + 1) % idx.length]; const c = x(i) * y(j) - x(j) * y(i); A += c; cx += (x(i) + x(j)) * c; cy += (y(i) + y(j)) * c; }
  A /= 2; cx /= 6 * A; cy /= 6 * A;
  let Ixx = 0, Iyy = 0, Ixy = 0;
  for (let k = 0; k < idx.length; k++) { const i = idx[k], j = idx[(k + 1) % idx.length]; const xi = x(i) - cx, yi = y(i) - cy, xj = x(j) - cx, yj = y(j) - cy; const c = xi * yj - xj * yi;
    Ixx += (yi * yi + yi * yj + yj * yj) * c; Iyy += (xi * xi + xi * xj + xj * xj) * c; Ixy += (xi * yj + 2 * xi * yi + 2 * xj * yj + xj * yi) * c; }
  Ixx /= 12; Iyy /= 12; Ixy /= 24;
  // covariance of x is Iyy/A, of y Ixx/A, cross Ixy/A
  const sxx = Iyy / A, syy = Ixx / A, sxy = Ixy / A;
  const tr = sxx + syy, det = sxx * syy - sxy * sxy; const l1 = tr / 2 + Math.sqrt(Math.max(0, tr * tr / 4 - det)), l2 = tr / 2 - Math.sqrt(Math.max(0, tr * tr / 4 - det));

    if (best === best && py - best > vh) vh = py - best;
  }
  return { H, crestY, lip: th[Math.floor(th.length / 2)], tip, area: Math.abs(A), LW: Math.sqrt(l1 / Math.max(1e-12, l2)), tilt: -ang, vh, outer: arc(p, 32, 64), under: arc(p, 64, 88) };
}
