# usage: savitsky.py <json>... ; clean-plane reference for each 0.2 s bin (a calculation, not a measurement).
# Savitsky (1964) flat plate, no deadrise: C_L0 = tau^1.1 (0.012 lam^0.5 + 0.0055 lam^2.5 / C_V^2), tau in degrees,
# lift = 0.5 rho V^2 b^2 C_L0, C_V = V / sqrt(g b), lam = wetted length / beam (held to the board's length).
# Drag = lift tan(tau) + friction, friction = 0.5 rho V^2 (lam b^2 / cos tau) C_f, C_f = 0.075 / (log10 Re - 2)^2 (ITTC 1957), Re = V lam b / nu.
# For each bin: the lift is the load the game's bottom pressure carries (its whole force, any direction); the speed is the board's;
# the trim is the one with least drag ("best"), and also the game's own angle of attack for a like-for-like line.
import json,sys,math
G=9.81; RHO=1025.0; NU=1.19e-6; LEN=1.778
def cl0(tau,lam,cv): return tau**1.1*(0.012*math.sqrt(lam)+0.0055*lam**2.5/cv**2)
def plate(L,V,b,tau):
  cv=V/math.sqrt(G*b); need=L/(0.5*RHO*V*V*b*b); lo,hi=1e-4,LEN/b
  if cl0(tau,hi,cv)<need: return None
  for _ in range(60):
    mid=(lo+hi)/2
    if cl0(tau,mid,cv)<need: lo=mid
    else: hi=mid
  lam=hi; re=max(V*lam*b/NU,1e4); cf=0.075/(math.log10(re)-2)**2
  fr=0.5*RHO*V*V*lam*b*b/math.cos(math.radians(tau))*cf
  return L*math.tan(math.radians(tau)), fr, lam
def best(L,V,b):
  out=None
  for i in range(10,300):
    tau=i/10; r=plate(L,V,b,tau)
    if r and (out is None or r[0]+r[1]<out[1]+out[2]): out=(tau,r[0],r[1],r[2])
  return out
for p in sys.argv[1:]:
  for o in json.load(open(p)):
    M=o['mass']['board']+o['mass']['rider']; W=M*G
    print(f"\n## {o['name']}")
    print(" t1   deg   v   | load(bw) attack | game: induced friction total (g) L/D | plate b=0.464 best: trim  lam  induced friction total (g) L/D | plate at game's attack: total (g) | plate b=0.32 best: trim total (g)")
    for b in o['bins']:
      P=math.sqrt(b['pressure_t']**2+b['pressure_n']**2+b['pressure_y']**2); V=b['v']
      if V<1.5 or P<50: continue
      gi=-b['pressure_t']; gf=-b['friction_t']; gt=gi+gf
      r=best(P,V,0.464); r2=best(P,V,0.32); s=plate(P,V,0.464,max(1.0,b['attack']))
      f=lambda x:f"{x/W:.3f}"
      print(f"{b['t1']:4.1f} {b['turned']:5.0f} {V:4.2f} | {P/W:5.2f}   {b['attack']:5.1f}  | {f(gi)} {f(gf)} {f(gt)}  {P/gt:4.1f} | "
            + (f"{r[0]:4.1f} {r[3]:4.2f} {f(r[1])} {f(r[2])} {f(r[1]+r[2])}  {P/(r[1]+r[2]):4.1f}" if r else "cannot carry it") + " | "
            + (f"{f(s[0]+s[1])} (lam {s[2]:.2f})" if s else "cannot carry it") + " | " + (f"{r2[0]:4.1f} {f(r2[1]+r2[2])}" if r2 else "cannot carry it"))
