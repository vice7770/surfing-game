function dist(ax: number, ay: number, bx: number, by: number) { return Math.hypot(ax - bx, ay - by); }
function segDist(px: number, py: number, ax: number, ay: number, bx: number, by: number) {
  const dx = bx - ax, dy = by - ay, l = dx * dx + dy * dy; let t = l > 0 ? ((px - ax) * dx + (py - ay) * dy) / l : 0; t = Math.max(0, Math.min(1, t));
  return dist(px, py, ax + t * dx, ay + t * dy);
}
function crossings(p: ArrayLike<number>, from = 0, to = PROFILE_POINTS - 1) {
  let n = 0;
  const x = (i: number) => p[2 * i], y = (i: number) => p[2 * i + 1];
  for (let i = from; i < to; i++) for (let j = i + 2; j < to; j++) {
    const ax = x(i), ay = y(i), bx = x(i + 1), by = y(i + 1), cx = x(j), cy = y(j), dx = x(j + 1), dy = y(j + 1);
    const d1 = (bx - ax) * (cy - ay) - (by - ay) * (cx - ax);
    const d2 = (bx - ax) * (dy - ay) - (by - ay) * (dx - ax);
    const d3 = (dx - cx) * (ay - cy) - (dy - cy) * (ax - cx);
    const d4 = (dx - cx) * (by - cy) - (dy - cy) * (bx - cx);
    if (d1 * d2 < 0 && d3 * d4 < 0) n++;
  }
  return n;
}
function measure(p: ArrayLike<number>) {
  const x = (i: number) => p[2 * i], y = (i: number) => p[2 * i + 1];
  const crestY = y(LANDMARK.crest);
  let minY = Infinity; for (let i = LANDMARK.crest; i < PROFILE_POINTS; i++) minY = Math.min(minY, y(i));
  const H = crestY - minY;
  const th: number[] = [];
  for (let i = 44; i <= 60; i++) { let m = Infinity; for (let j = 68; j < 88; j++) m = Math.min(m, segDist(x(i), y(i), x(j), y(j), x(j + 1), y(j + 1))); th.push(m); }
  th.sort((a, b) => a - b);
  let tip = Infinity; for (let i = 56; i <= 62; i++) for (let j = 66; j < 74; j++) tip = Math.min(tip, segDist(x(i), y(i), x(j), y(j), x(j + 1), y(j + 1)));
  let a = 0; for (let i = 64; i < 112; i++) a += x(i) * y(i + 1) - x(i + 1) * y(i); a += x(112) * y(64) - x(64) * y(112);
  // vertical void: max over underside points 64..88 of (y - surface beneath among 88..127)
  let vh = 0;
  for (let i = 64; i <= 88; i++) {
    const px = x(i), py = y(i); let best = NaN;
    for (let k = 88; k < 127; k++) { const x0 = x(k), x1 = x(k + 1); if (x0 === x1 || (x0 - px) * (x1 - px) > 0) continue; const u = y(k) + (px - x0) / (x1 - x0) * (y(k + 1) - y(k)); if (u < py && !(u <= best)) best = u; }
    if (best === best && py - best > vh) vh = py - best;
  }
  return { H, lipMedian: th[Math.floor(th.length / 2)], lipMax: th[th.length - 1], tip, area: Math.abs(a) / 2, vh };
}
