function reset(v: Variant) {
  foamKeys.forEach((k, i) => v.scene.foam[k].set(v.initial.foam[i] as never));
  v.scene.foam.sourceCount = v.initial.sourceCount; v.scene.foam.breakingCount = v.initial.breakingCount;
  v.scene.aeration.air.set(v.initial.air[0]); v.scene.aeration.depth.set(v.initial.air[1]); v.scene.aeration.turbulence.set(v.initial.air[2]);
  v.foam.nextDense.fill(0); v.foam.nextResidual.fill(0);
  v.air.nextAir.fill(0); v.air.nextDepth.fill(0); v.air.nextTurbulence.fill(0); v.air.stirred.fill(0);

}
function run(v: Variant): number[] {
  const t0 = performance.now();

  else (v.scene as SurfZoneSimulation).foam.updateGrid(plan.dt, v.scene.whitewaterStrength, v.stencil as AdvectionStencil);
  const t1 = performance.now();

  const t2 = performance.now();

  else (v.scene as SurfZoneSimulation).aeration.update(plan.dt, v.stencil as AdvectionStencil);
  const t3 = performance.now();
  v.scene.writeUniformSnapshot(v.outputs[0], v.outputs[1], v.outputs[2], input.grid);
  const t4 = performance.now();

}
function stable(v: Variant) {
  const s = v.scene.solver, ss = v.stencil as unknown as StencilState;
  solverKeys.forEach((k, i) => { if (s[k] !== v.identities.solver[i] || sha(bytes(s[k])) !== v.solverHashes[i]) throw Error('Solver identity/bytes changed:' + v.name + '.' + k); });
  if (s.time !== v.time || v.scene.whitewaterStrength !== v.identities.strength || sha(bytes(v.scene.whitewaterStrength)) !== v.strengthSha) throw Error('Clock/source strength changed');
  if (v.foam.windowX !== v.windowX[0] || v.air.windowX !== v.windowX[1]) throw Error('Held field window changed');
  foamKeys.forEach((k, i) => { if (v.scene.foam[k] !== v.identities.foam[i]) throw Error('Public foam identity changed:' + k); });
  airKeys.forEach((k, i) => { if ((v.scene.aeration as unknown as Record<string, ArrayBufferView>)[k] !== v.identities.air[i]) throw Error('Air public/private identity changed:' + k); });
  if (v.foam.nextDense !== v.identities.nextFoam[0] || v.foam.nextResidual !== v.identities.nextFoam[1]) throw Error('Private foam identity changed');
  if (ss.ready || ss.solver !== s || ss.xCenters !== s.xCenters || ss.zCenters !== s.zCenters || ss.dt !== plan.dt || ss.time !== s.time
    || ss.xMin !== s.xCenters[0] || ss.zMin !== s.zCenters[0] || ss.dx !== s.dx || ss.nx !== s.nx || ss.nz !== s.nz) throw Error('Consumed shared-stencil descriptor differs');
}

}
