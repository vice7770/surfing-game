def archive(source, relative, encoding='identity', expanded=None, original_aliases=()):
    original = source.read_bytes()
    if encoding == 'gzip-mtime0':
        buffer = io.BytesIO()
        with gzip.GzipFile(filename='', mode='wb', fileobj=buffer, mtime=0, compresslevel=9) as stream:
            stream.write(original)
        stored = buffer.getvalue()
        assert stored[4:8] == b'\0\0\0\0'
        assert gzip.decompress(stored) == original
        expanded = original
    elif encoding == 'retained-original-gzip':
        stored = original
        assert expanded is not None and gzip.decompress(stored) == expanded
    else:
        assert encoding == 'identity'
        stored = original
        expanded = original
    target = DEST / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(stored)
    assert target.read_bytes() == stored
    record = {
        'storedPath': relative, 'encoding': encoding,
        'storedBytes': len(stored), 'storedSha256': sha(stored),
        'expandedBytes': len(expanded), 'expandedSha256': sha(expanded),
        'original': describe(source),
        'originalAliases': [describe(Path(p)) for p in original_aliases],
    }
    for alias in record['originalAliases']:
        assert Path(alias['originalPath']).read_bytes() == original
    payloads.append(record)
    return record

    return record
