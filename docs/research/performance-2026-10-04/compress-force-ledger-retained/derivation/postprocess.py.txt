def get(record,key):
    assert record[0]=='r'
    return record[indexes[record[1]][key]]
def maybe(record,key,default=None):
    assert record[0]=='r'
    index=indexes[record[1]].get(key)
    return record[index] if index is not None else default
def array(token):
    assert token[0]=='a'
    return token[1:]
def decode(token):
    if not isinstance(token,list):return token
    tag=token[0]
    if tag=='a':return [decode(x) for x in token[1:]]
    if tag=='r':return {fields[field]:decode(token[i+2]) for i,field in enumerate(shapes[token[1]])}
    if tag=='u':return {'undefined':True}
    if tag=='n':return {'-0':-0.0,'NaN':float('nan'),'+Infinity':float('inf'),'-Infinity':-float('inf')}[token[1]]
    raise ValueError(f'Unknown schema tag {tag}')
def turning(values,deadband=.2):
    high=low=values[0]; high_at=low_at=0; direction=0; points=[]
    for index,value in enumerate(values):
        if direction>=0 and value>high:high=value;high_at=index
        if direction<=0 and value<low:low=value;low_at=index
        if direction>=0 and high-value>deadband:
            points.append({'index':high_at,'value':high});direction=-1;low=value;low_at=index
        elif direction<=0 and value-low>deadband:
            points.append({'index':low_at,'value':low});direction=1;high=value;high_at=index
    return {'points':points,'largestSwing':max([abs(points[i]['value']-points[i-1]['value']) for i in range(1,len(points))]+[0])}
def flips(values):
    last=0; total=0
    for value in values:
        if abs(value)<.05:continue
        sign=1 if value>0 else -1
        if last and sign!=last:total+=1
        last=sign
    return total
def magnitude(values):return math.sqrt(sum(x*x for x in values))
