import { test } from 'node:test';
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { WORK, assertInputs } from './canonical-resolver.mjs';
const read = (arm, file) => readFileSync(resolve(WORK, arm, 'src', file), 'utf8');
test('original slope sampler and GLSL remain literally unchanged', () => {
  assertInputs();
  const old = read('baseline', 'scene/water/cubicSurface.ts');
  const next = read('candidate', 'scene/water/cubicSurface.ts');
  const helper = next.indexOf('/**\n * Height-only CPU mirror');
  const glsl = '/** GLSL twin of `sampleCubicSurface`';
  assert.equal(next.slice(0, helper), old.slice(0, old.indexOf(glsl)));
  assert.equal(next.slice(next.indexOf(glsl)), old.slice(old.indexOf(glsl)));
  const onlyHelper = next.slice(helper, next.indexOf(glsl));
  for (const axis of ['x', 'z']) {
    const t = `t${axis}`;
    for (const expr of [`(-${t}3 + 2 * ${t}2 - ${t}) / 2`, `(3 * ${t}3 - 5 * ${t}2 + 2) / 2`, `(-3 * ${t}3 + 4 * ${t}2 + ${t}) / 2`, `(${t}3 - ${t}2) / 2`]) assert.ok(onlyHelper.includes(expr));
  }
  for (const expr of ['for (let j = 0; j < 4; j += 1)', 'for (let i = 0; i < 4; i += 1)', 'Math.min(grid.nz - 1, Math.max(0, j0 + j - 1))', 'Math.min(grid.nx - 1, Math.max(0, i0 + i - 1))', 'data[(row * grid.nx + column) * 2]', 'height += wz * wx * value']) assert.ok(onlyHelper.includes(expr));
});
test('Rich callback change is the only caller change', () => {
  const old = read('baseline', 'scene/barrel/SweptBarrel.ts');
  const next = read('candidate', 'scene/barrel/SweptBarrel.ts');
  assert.equal(next, old.replace("import { sampleCubicSurface } from '../water/cubicSurface';", "import { sampleCubicHeight } from '../water/cubicSurface';")
    .replace('sampleCubicSurface(surfaceData, grid, x, z).height', 'sampleCubicHeight(surfaceData, grid, x, z)'));
});
