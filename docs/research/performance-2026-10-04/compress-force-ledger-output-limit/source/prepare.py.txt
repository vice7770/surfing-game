import hashlib, json, pathlib, re, subprocess

sha = lambda data: hashlib.sha256(data).hexdigest()
entries = ['src/physics/AttachedRider.ts', 'src/physics/BoardBody.ts', 'src/physics/PlaneWater.ts', 'src/physics/hullForces.ts']
seen, stack = set(), [root / x for x in entries]
records = []
while stack:
    path = stack.pop().resolve()
    if path in seen:
        continue
    seen.add(path)
    data = path.read_bytes()
    records.append({'path': str(path), 'role': 'conservative-runtime-import-closure', 'bytes': len(data), 'sha256': sha(data)})
    if path.suffix not in ['.ts', '.js', '.mjs']:
        continue
    for line in data.decode().splitlines():
        if not line.startswith(('import ', 'export ')) or line.startswith(('import type ', 'export type ')):
            continue
        match = re.search(r'''(?:from\s+|^import\s*)['"]([^'"]+)['"]''', line)
        if not match or not match[1].startswith('.'):
            continue
        base = path.parent / match[1]
        candidates = [base, pathlib.Path(str(base) + '.ts'), pathlib.Path(str(base) + '.json'), base / 'index.ts']
        dependency = next((p for p in candidates if p.is_file()), None)
        if dependency is None:
            raise RuntimeError(f'Unresolved source authority {path}: {match[1]}')
        stack.append(dependency)
