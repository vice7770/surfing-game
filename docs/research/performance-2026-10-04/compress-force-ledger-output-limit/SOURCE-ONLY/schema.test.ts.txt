import { describe, expect, it } from 'vitest';
import { decodeReport, encodeReport, failureSummary } from './schema';

describe('output-only compact schema (unexecuted synthetic preparation)', () => {
  it('losslessly round-trips changing/optional nested record shapes, exact numbers, events and window order', () => {
    const input = { rows: [{ dt: 1 / 60, force: [1, -0, NaN], marker: undefined }, { force: [Infinity, -Infinity, 1e-200], event: 'first tip' }], windows: [3, 1, 4], metadata: { arbitraryArray: ['r', 1, ['n', 'NaN']], text: '__proto__', empty: {} } };
    const parsed = JSON.parse(JSON.stringify(encodeReport(input)));
    expect(decodeReport(parsed)).toEqual(input);
    expect(Object.is(decodeReport(parsed).rows[0].force[1], -0)).toBe(true);
    expect(Object.hasOwn(decodeReport(parsed).rows[0], 'marker')).toBe(true);
  });
  it('stores repeated field names once and reduces bytes on a declared synthetic repeated row shape', () => {
    const rows = Array.from({ length: 1000 }, (_, i) => ({ referenceBankBefore: i / 100, referenceBankAfter: i / 100 + 1, compressTime: i / 60, requestedImpulse: [i, 1, 3], projectedImpulse: [i / 2, 1, 3], riderWorkDelta: { assist: i - 3, carry: i + 2, leanOut: 0 }, contactFeasible: i % 2 === 0 }));
    const input = { rows };
    const encoded = encodeReport(input);
    expect(encoded.fields.filter(x => x === 'referenceBankBefore')).toHaveLength(1);
    expect(Buffer.byteLength(JSON.stringify(encoded))).toBeLessThan(Buffer.byteLength(JSON.stringify(input)) * 0.7);
    expect(decodeReport(JSON.parse(JSON.stringify(encoded)))).toEqual(input);
  });
  it('preserves own __proto__ data without prototype mutation and rejects malformed tuples', () => {
    const input = JSON.parse('{"__proto__":{"x":1},"constructor":2}');
    const decoded = decodeReport(encodeReport(input));
    expect(Object.getPrototypeOf(decoded)).toBe(Object.prototype); expect(Object.hasOwn(decoded, '__proto__')).toBe(true);
    expect(decoded.__proto__).toEqual({ x: 1 });
    expect(() => decodeReport({ edition: 'compress-ledger-tuples-v1', fields: ['x'], shapes: [[0]], payload: ['r', 0] })).toThrow('length');
  });
  it('retains usable partial verified state/events but never upgrades an original failed report', () => {
    const input = { valid: false, error: 'parity at step50', paritySteps: 50, observedSubsteps: 1601,
      latestVerified: { speed: 7, variant: 'compress', step: 49, bank: -0 },
      cases: [{ speed: 8, variant: 'held', paritySteps: 108, observedSubsteps: 3456, events: [{ kind: 'first tip', t: 0.2 }], legacyYawTurningPoints: { points: [{ index: 3, value: 1 }], largestSwing: 0 }, signFlips: 0 }] };
    const summary = failureSummary(input, 'serialization bound', 90_000_000);
    expect(summary.valid).toBe(false); expect(summary.originalReportValid).toBe(false);
    expect(summary.originalReportError).toBe('parity at step50'); expect(summary.paritySteps).toBe(50);
    expect(summary.completedCases[0].events).toEqual(input.cases[0].events);
    expect(decodeReport(JSON.parse(JSON.stringify(encodeReport(summary)))).latestVerified).toEqual(input.latestVerified);
  });
});
