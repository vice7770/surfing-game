import { readFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
export const WORK = '/private/tmp/surf-spray-interleaved-20261004';
export const sha = raw => createHash('sha256').update(raw).digest('hex');
export function assertReady() {
  const raw = readFileSync(WORK + '/ready.json'), ready = JSON.parse(raw);
  if (ready.schema !== 'spray-interleaved-source-ready/v1' || ready.executed) throw Error('Wrong ready authority');
  for (const pin of ready.inputs) {
    const bytes = readFileSync(pin.path);
    if (bytes.length !== pin.bytes || sha(bytes) !== pin.sha256) throw Error('Changed input: ' + pin.path);
  }
  return { ready, sha256: sha(raw) };
}
