def digest(raw):
    return sha256(raw).hexdigest()

def write_json(path, value):
    path.write_text(json.dumps(value, indent=2) + '\n')

payloads = []
by_content = {}
alias_lookup = {}

def add(source, relative, expected=None):
    raw = source.read_bytes()
    pin = {'path': str(source), 'bytes': len(raw), 'sha256': digest(raw)}
    if expected is not None:
        assert all(pin[k] == expected[k] for k in ('path', 'bytes', 'sha256')), source
    identity = (pin['bytes'], pin['sha256'])
    if identity in by_content:
        entry = by_content[identity]
    else:
        relative = str(relative)
        if relative.endswith(('.ts', '.js', '.mjs', '.py')):
            relative += '.txt'
        compressed = len(raw) >= 1200 and not relative.endswith('.log')
        if compressed:
            relative += '.gz'
        stored = gzip.compress(raw, compresslevel=6, mtime=0) if compressed else raw
        target = DEST / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(stored)
        entry = {'storedPath': relative, 'encoding': 'gzip-generated-mtime0' if compressed else 'identity',
                 'storedBytes': len(stored), 'storedSha256': digest(stored),
                 'expandedBytes': len(raw), 'expandedSha256': digest(raw), 'originalAliases': []}
        by_content[identity] = entry
        payloads.append(entry)
    if pin not in entry['originalAliases']:
        entry['originalAliases'].append(pin)
    alias_lookup[(pin['path'], pin['bytes'], pin['sha256'])] = entry
    return pin

def archived_reference(original, role):
    record = live_alias[(original['path'], original['bytes'], original['sha256'])]
    source = live_base / record['storedPath']
    stored = source.read_bytes()
    assert len(stored) == record['storedBytes'] and digest(stored) == record['storedSha256']
    raw = gzip.decompress(stored) if record['encoding'].startswith('gzip') else stored
    assert len(raw) == original['bytes'] and digest(raw) == original['sha256']
    external.append({'originalPath': original['path'], 'bytes': original['bytes'], 'sha256': original['sha256'],
                     'verification': {'kind': 'archive', 'repositoryRelativePath': str(source.relative_to(REPO)),
                                      'encoding': record['encoding'], 'storedBytes': record['storedBytes'],
                                      'storedSha256': record['storedSha256']}, 'role': role})
    return raw
