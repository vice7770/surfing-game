import { mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { performance } from 'node:perf_hooks';
import { SessionCDP } from './session-cdp.mjs';
import { attribute } from './attribution.mjs';

const wait = ms => new Promise(resolve => setTimeout(resolve, Math.max(0, ms)));
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
const stamp = () => ({ iso: new Date().toISOString(), monoMs: performance.now() });
const bounded = async promise => {
  let timer;
  try{return await Promise.race([promise,new Promise((_,reject)=>{timer=setTimeout(()=>reject(Error('Page observer command deadline')),10000);})]);}
  finally{clearTimeout(timer);}
};

const witness = `(() => {
  const d=window.breaklineDiagnostics, host=d?.mode?.host, worker=host?.port;
  return { rendererNow:performance.now(),screen:document.querySelector('#app')?.dataset.screen,
    ready:d?.mode?.ready===true,config:d?.mode?.config,status:host?.snapshot?.status,
    workerUrl:worker?.__profileUrl,workerStart:worker?.__profileStart,
    constructorObserved:worker?.__profileObserved===true,
    graphics:JSON.parse(localStorage.getItem('breakline.settings.v1')??'{}').graphics,
    viewport:innerWidth+' × '+innerHeight,browserDpr:devicePixelRatio,
    canvas:(()=>{const c=document.querySelector('canvas');return c?c.width+' × '+c.height:'';})(),
    waterLook:d?.water?.drawnLook,vertexNormals:d?.water?.vertexNormals,particles:d?.mode?.particleLevel,
    renderSpacing:d?.water?.grid?.spacing,maxBatchSteps:host?.maxBatchSteps};
})()`;

export async function readyWitness(page, ordinary, workerUrl) {
  const seen = await bounded(page.eval(witness));
  const failures = [];
  if (!seen.ready || seen.screen !== 'ride' || seen.status?.compute !== 'gpu') failures.push('Ordinary ride is not ready on GPU');
  if (!seen.constructorObserved || seen.workerUrl !== workerUrl || !seen.workerStart?.rider) failures.push('Actual host worker identity/start differs');
  for (const [key, value] of Object.entries(ordinary.expectedConfig)) {
    if (seen.config?.[key] !== value || seen.workerStart?.config?.[key] !== value) failures.push('Config differs: '+key);
  }
  for (const [key, value] of Object.entries(ordinary.graphics)) if (seen.graphics?.[key] !== value) failures.push('Graphics differs: '+key);
  if (seen.viewport !== ordinary.expectedViewport || seen.browserDpr !== ordinary.expectedBrowserDpr || seen.canvas !== ordinary.expectedCanvas
    || seen.waterLook !== 'rich' || seen.vertexNormals !== false || seen.particles !== 'high' || seen.renderSpacing !== 2 || seen.maxBatchSteps !== 1) failures.push('Native quality/display/worker cadence differs');
  if (failures.length) throw Error(failures.join('; '));
  return seen;
}

export async function collectLateProfiles({ page, ordinary, workerUrl, port, out, origin, sources, provenance }) {
  mkdirSync(out, { recursive: true });
  const report = { kind:'JS CPU attribution diagnostic; not passive FPS acceptance',valid:false,origin,
    schedule:{waitFromOriginMs:75000,profileMs:8000,samplingIntervalUs:1000},provenance,commands:[],errors:[],profiles:{},startedAt:stamp() };
  const retain = () => writeFileSync(out+'/report.json', JSON.stringify(report, null, 2)+'\n');
  retain();
  let cdp;
  const sessions = [], started = new Set(), stopped = new Set();
  const command = async (session, method, params={}) => {
    const row={context:session.name,method,sent:stamp()}; report.commands.push(row);retain();
    try {const value=await cdp.send(method,params,session.id);row.received=stamp();retain();return value;}
    catch(error){row.error=String(error);row.received=stamp();retain();throw error;}
  };
  const stop = async session => {
    if (!started.has(session.name) || stopped.has(session.name)) return;
    stopped.add(session.name);
    const {profile}=await command(session,'Profiler.stop');
    const path=out+'/'+session.name+'.cpuprofile';
    const bytes=Buffer.from(JSON.stringify(profile));writeFileSync(path,bytes);
    report.profiles[session.name]={path,sha256:hash(bytes),bytes:bytes.length,samples:profile.samples?.length??0,
      startTime:profile.startTime,endTime:profile.endTime};retain();
    if (!(profile.samples?.length>0) || profile.samples.length!==profile.timeDeltas?.length) throw Error('Missing or malformed CPU samples');
    writeFileSync(out+'/'+session.name+'-attribution.json',JSON.stringify(attribute(profile,sources),null,2)+'\n');
  };
  try {
    // This timer starts at the ready-guarded canonical sample entry, after the original five-second warmup.
    await wait(origin.monoMs+75000-performance.now());
    report.before=await readyWitness(page,ordinary,workerUrl);retain();
    const version=await(await fetch('http://127.0.0.1:'+port+'/json/version',{signal:AbortSignal.timeout(5000)})).json();
    report.browserVersion=version.Browser;cdp=await SessionCDP.connect(version.webSocketDebuggerUrl);
    const frame=(await bounded(page.send('Page.getFrameTree'))).frameTree.frame;
    const {targetInfos}=await cdp.send('Target.getTargets');report.targetsBefore=targetInfos;
    const pages=targetInfos.filter(row=>row.type==='page'&&row.url===provenance.url);
    const workers=targetInfos.filter(row=>row.type==='worker'&&row.url===workerUrl&&row.parentFrameId===frame.id);
    if(pages.length!==1||workers.length!==1)throw Error('Ambiguous/missing exact page or parent-frame worker target');
    report.frame=frame;report.worker=workers[0];report.pageTarget=pages[0];retain();
    for(const [name,target]of [['page',pages[0]],['worker',workers[0]]]){
      const {sessionId}=await cdp.send('Target.attachToTarget',{targetId:target.targetId,flatten:true});
      sessions.push({name,id:sessionId,targetId:target.targetId});
    }
    report.sessions=sessions;retain();
    for(const session of sessions){await command(session,'Profiler.enable');await command(session,'Profiler.setSamplingInterval',{interval:1000});}
    const starts=await Promise.allSettled(sessions.map(async session=>{await command(session,'Profiler.start');started.add(session.name);}));
    for(const result of starts)if(result.status==='rejected')throw result.reason;
    report.recordingStarted=stamp();retain();
    await wait(8000);
    const stops=await Promise.allSettled(sessions.map(stop));
    for(const result of stops)if(result.status==='rejected')throw result.reason;
    report.after=await readyWitness(page,ordinary,workerUrl);
    report.targetsAfter=(await cdp.send('Target.getTargets')).targetInfos;
    if(!report.targetsAfter.some(row=>row.targetId===report.worker.targetId&&row.url===workerUrl&&row.parentFrameId===frame.id))throw Error('Worker target replaced during sample');
    const rows=await bounded(page.eval('window.__perf.snapshots'));
    report.pipelineWitnesses=rows.filter(row=>row.wall>=report.before.rendererNow&&row.wall<=report.after.rendererNow);
    if(!(report.after.status.seaTime>report.before.status.seaTime))throw Error('Physics did not advance during profile');
    report.valid=true;
  }catch(error){report.errors.push(String(error.stack??error));}
  finally{
    if(cdp){
      for(const session of sessions){
        try{await stop(session);}catch(error){report.errors.push('Partial stop: '+String(error));}
        try{await cdp.send('Target.detachFromTarget',{sessionId:session.id});}catch(error){report.errors.push('Detach: '+String(error));}
      }
      cdp.close();
    }
    report.finishedAt=stamp();retain();
  }
  return report;
}
