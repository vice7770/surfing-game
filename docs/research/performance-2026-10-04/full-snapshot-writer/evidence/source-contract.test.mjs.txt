import { readFileSync } from 'node:fs';
import { strict as assert } from 'node:assert';
import test from 'node:test';
const baseline = readFileSync(new URL('./baseline/src/wave/SurfZoneSimulation.ts', import.meta.url), 'utf8');
const candidate = readFileSync(new URL('./candidate/src/wave/SurfZoneSimulation.ts', import.meta.url), 'utf8');
function method(source, signature) {
  const start = source.indexOf(signature); assert(start >= 0);
  const open = source.indexOf('{', start); let depth = 1, end = open + 1;
  while (depth && end < source.length) { if (source[end] === '{') depth++; else if (source[end] === '}') depth--; end++; }
  assert.equal(depth, 0); return { start, end, body: source.slice(start, end) };
}
test('only the required full method changes; public generic optional method stays literal', () => {
  const a = method(baseline, '  writeUniformSnapshot('), b = method(candidate, '  writeUniformSnapshot(');
  assert.equal(baseline.slice(0, a.start), candidate.slice(0, b.start));
  assert.equal(baseline.slice(a.end), candidate.slice(b.end));
  assert.equal(method(baseline, '  private writeUniformFields(').body, method(candidate, '  private writeUniformFields(').body);
  const rebindings = JSON.parse(readFileSync(new URL('./qa-rebindings.json', import.meta.url), 'utf8'));
  for (const [arm, source] of [['baseline', baseline], ['candidate', candidate]]) {
    let expected = source;
    for (const row of rebindings.records.filter(row => row.arm === arm)) expected = expected.replaceAll("from '" + row.relative + "'", "from '" + row.canonical + "'");
    assert.equal(readFileSync(new URL('./qa/' + arm + '.ts', import.meta.url), 'utf8'), expected);
  }
});
test('full path retains literal coefficients/stores and dynamic void calls without optional channel dispatch', () => {
  const source = method(candidate, '  writeUniformSnapshot(').body;
  const generic = method(baseline, '  private writeUniformFields(').body;
  for (const prefix of ['const w00', 'const w10', 'const w01', 'const w11', 'const bottom', 'surface[o] =', 'surface[o + 1] =', 'flow[o] =', 'flow[o + 1] =', 'aeration[o] =', 'aeration[o + 1] =']) {
    const statement = body => body.slice(body.indexOf(prefix), body.indexOf(';', body.indexOf(prefix)) + 1).replace(/\s+/g, ' ');
    assert.equal(statement(source), statement(generic));
  }
  assert.equal((source.match(/this\.aeration\.voidFraction\(i\)/g) ?? []).length, 1);
  assert(!/if \((surface|flow|aeration)([ )]| &&)/.test(source));
  assert(!source.includes('surface || flow ?'));
});
