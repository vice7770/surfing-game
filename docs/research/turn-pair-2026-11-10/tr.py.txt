# usage: tr.py <json> <t0> <t1> [keys] ; prints the trace's frames between t0 and t1
import json,sys
d=json.load(open(sys.argv[1]))['traces']; t0=float(sys.argv[2]); t1=float(sys.argv[3])
keys=(sys.argv[4] if len(sys.argv)>4 else 't,ph,st,cp,frac,ahead,ff,rate,v,fin,bank,ref,bkr,arest,sw,roll,load,lim,leg,lo,as,ca,dr').split(',')
for name,fr in d.items():
  print(name); print(' '.join(f'{k:>6s}' for k in keys))
  for f in fr:
    if t0<=f['t']<=t1: print(' '.join(f'{str(f[k]):>6s}' for k in keys))
