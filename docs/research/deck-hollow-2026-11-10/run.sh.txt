# Deck hollow study: every run of ../carve-carry-2026-11-10/run.sh.txt, once with the switch off and once on (HOLLOW is a scratch env override in BoardBody).
T=$1
for h in off on; do
for c in 0 0.2 0.4; do
  export CARRY=$c HOLLOW=$h
  pair() { DRIVES=off CLIMBS=hold npx vitest run src/physics/zz-turnpair.test.ts >/dev/null 2>&1; }
  flat() { npx vitest run src/physics/zz-carryflat.test.ts >/dev/null 2>&1; }
  for d in 20 35; do SECONDS=16 WAIT=1 LOW=0.25 DROP=$d TOPS=60/0.5,60/0.7 OUT=$T/pairB-$d-c$c-$h.json pair; done
  SECONDS=16 WAIT=1 LOW=0.25 DROP=20 TOPS=60/0.3,45/0.3 OUT=$T/pairC-c$c-$h.json pair
  PLAN=cutback CUTS=150,180 ATS=0.5,0.75 OUT=$T/cut-c$c-$h.json pair
  PLAN=cutback CUTS=90,120 OUT=$T/cut2-c$c-$h.json pair
  PLAN=arc OUT=$T/arc-c$c-$h.json flat; PLAN=arc SPEEDS=9 OUT=$T/arc9-c$c-$h.json flat
  PLAN=trim SPEEDS=0,1.5 OUT=$T/trim-c$c-$h.json flat
  echo done $h $c
done; done
