const library = () => new ProfileLibrary([tubeCase(0.3)]);

/** A closed face/underside/top strip. Its three horizontal crossings are y=0,2,3 at z=1. */
function cavity(slices = 2): LoftResult {
  const blank = new SweptLoft(library(), 0.05).build(new Float32Array(), 0, 0, () => 0);
  const out = { ...blank };
  for (const [name, value] of Object.entries(blank)) {
    if (!ArrayBuffer.isView(value)) continue;
    const size = name === 'indices' ? 6 * (S - 1) * (slices - 1)
      : name === 'positions' || name === 'normals' ? 3 * S * slices : name.startsWith('slice') ? slices : S * slices;
    (out as unknown as Record<string, unknown>)[name] = value instanceof Uint8Array ? new Uint8Array(size)
      : value instanceof Uint32Array ? new Uint32Array(size) : new Float32Array(size);
  }
  out.sliceCount = slices; out.vertexCount = slices * S; out.indexCount = 6 * (S - 1) * (slices - 1);
  out.sliceFront.fill(1); out.sliceRayZ.fill(1); out.sliceWeight.fill(1); out.sliceLife.fill(0.3);
  const profile = [[0, 0], [2, 0], [2, 2], [0, 2], [0, 3], [2, 3]];
  for (let s = 0; s < slices; s += 1) {
    out.sliceSigma[s] = s;
    for (let j = 0; j < S; j += 1) {
      const [z, y] = profile[Math.min(j, profile.length - 1)], v = s * S + j;
      out.positions.set([s, y, z], 3 * v); out.normals[3 * v + 1] = 1;
    }
    if (s + 1 < slices) {
      out.sliceJoined[s] = 1;
      for (let j = 0; j + 1 < S; j += 1) {
        const v = s * S + j;
        out.indices.set([v, v + S, v + 1, v + 1, v + S, v + S + 1], 6 * ((S - 1) * s + j));
      }
    }
  }
  return out;
}

/** Independent production contact index/triangle/parity code on the synthetic geometry. */
function contactFor(loft: LoftResult) {
  const contact = new SweptContact(library(), 0.05, { bucket: Infinity });
  vi.spyOn((contact as unknown as { loft: SweptLoft }).loft, 'build').mockReturnValue(loft);
  contact.update(new Float32Array(), 0, 0, () => 0);
  return contact;
}

type Fixture = { waterGrid: SurfaceGrid; surfaceData: number[]; front: (number | null)[]; frontCount: number;
  config: { tide: number }; status: SurfZoneStatus };
const cases = () => libraryFromBytes(barrelCasesFor('padang').map(c =>
  new Uint8Array(readFileSync(new URL('../../../public/' + c.asset, import.meta.url)))));
function fixture(at: number) {
  const data: Fixture = JSON.parse(readFileSync(new URL(`../../../docs/research/tube-stability-2026-10-03/fixtures/at-${at}.json`, import.meta.url), 'utf8'));
  const front = Float32Array.from(data.front, v => v ?? Number.NaN), surface = Float32Array.from(data.surfaceData);
  const heightAt = (x: number, z: number) => sampleCubicSurface(surface, data.waterGrid, x, z).height;
  return { data, front, surface, heightAt };
}

describe('bounded rendered cavity air', () => {
  it('requires the locally shaded opaque mask, retaining zero and dither-band fallback despite global activity', () => {
    const grid = { xMin: 0, zMin: 0, spacing: 1, nx: 2, nz: 2 };
    const mask = new Uint8Array([255, 255, 255, 255]);
    expect(opaqueBarrelColumn(mask, grid, 0.3, 0.5)).toBe(true);
    mask[3] = 0;
    expect(opaqueBarrelColumn(mask, grid, 0.3, 0.5)).toBe(false);
    mask[3] = 254;
    expect(opaqueBarrelColumn(mask, grid, 0.3, 0.5)).toBe(false);
    expect(opaqueBarrelColumn(mask, grid, 0, 0)).toBe(true); // Zero-weight neighbours do not shade this node.
    expect(opaqueBarrelColumn(mask, grid, -0.1, 0.5)).toBe(false);
  });
  it('matches independent contact air, floor/roof water and half-open ray/triangle ties', () => {
    const loft = cavity(3), query = new RenderedCavityQuery(), contact = contactFor(loft), hit = createContactHit();
    for (const x of [0, 0.3, 0.5, 1, 1.5]) for (const y of [-0.1, 0, 1, 2, 2.5, 3.5]) {
      expect(contact.query(x, y, 1, hit)).toBe(true);
      const boundedAir = !hit.inWater && Number.isFinite(hit.ceilingY) && hit.floorY < y && y < hit.ceilingY;
      expect(query.contains(loft, x, y, 1)).toBe(boundedAir);
    }
    expect(query.contains(loft, 2, 1, 1)).toBe(false); // Last ray belongs to no half-open strip.
    expect(query.contains(loft, -0.1, 1, 1)).toBe(false);
  });

  it('never mistakes flat water, air under all crossings, or air above the roof for a cavity', () => {
    const loft = cavity(), query = new RenderedCavityQuery();
    expect(query.contains(loft, 0.3, -1, 1)).toBe(false);
    expect(query.contains(loft, 0.3, 4, 1)).toBe(false);
    for (let s = 0; s < 2; s += 1) for (let j = 0; j < S; j += 1) {
      loft.positions[3 * (s * S + j) + 1] = 0;
      loft.positions[3 * (s * S + j) + 2] = j;
    }
    expect(query.contains(loft, 0.3, -0.2, 1.2)).toBe(false);
    expect(query.contains(loft, 0.3, 0.2, 1.2)).toBe(false);
  });

  it('recomputes reused mutable geometry without a stale held query result', () => {
    const loft = cavity(), query = new RenderedCavityQuery();
    expect(query.contains(loft, 0.3, 1, 1)).toBe(true);
    for (let v = 0; v < loft.vertexCount; v += 1) loft.positions[3 * v + 1] += 2;
    expect(query.contains(loft, 0.3, 1, 1)).toBe(false);
    expect(query.contains(loft, 0.3, 3, 1)).toBe(true);
    loft.indexCount = 0;
    expect(query.contains(loft, 0.3, 3, 1)).toBe(false);
  });

  it('rejects overlapping fronts even where physical contact selects its first front', () => {
    const loft = cavity(4), query = new RenderedCavityQuery();
    loft.sliceJoined[1] = 0; loft.sliceFront[2] = loft.sliceFront[3] = 2;
    for (let v = 2 * S; v < 4 * S; v += 1) loft.positions[3 * v] -= 2;
    loft.indices.copyWithin(6 * (S - 1), 12 * (S - 1)); loft.indexCount = 12 * (S - 1);
    const contact = contactFor(loft), hit = createContactHit();
    expect(contact.query(0.3, 1, 1, hit)).toBe(true); expect(hit.inWater).toBe(false);
    expect(query.contains(loft, 0.3, 1, 1)).toBe(false);
  });

  it('retains the fallback for malformed indices, nonfinite geometry, query-edge degeneracy and overflow', () => {
    const query = new RenderedCavityQuery();
    const malformed = cavity(); malformed.indices[0] = 1;
    expect(query.contains(malformed, 0.3, 1, 1)).toBe(false);
    const nonfinite = cavity(); nonfinite.positions[7] = Number.NaN;
    expect(query.contains(nonfinite, 0.3, 1, 1)).toBe(false);
    expect(query.contains(cavity(), 0.3, 1, 2)).toBe(false);
    const overflow = cavity();
    for (let s = 0; s < 2; s += 1) for (let j = 0; j < S; j += 1) {
      overflow.positions[3 * (s * S + j) + 1] = j;
      overflow.positions[3 * (s * S + j) + 2] = 2 * (j % 2);
    }
    expect(S - 1).toBeGreaterThan(CONTACT.crossings);
    expect(query.contains(overflow, 0.3, 1, 1)).toBe(false);
    const extra = cavity();
    for (let s = 0; s < 2; s += 1) {
      extra.positions[3 * (s * S + 6) + 2] = 0;
      extra.positions[3 * (s * S + 6) + 1] = 4;
      extra.positions[3 * (s * S + 7) + 1] = 5;
    }
    expect(query.contains(extra, 0.3, 1, 1)).toBe(false);
  });
});

describe('fresh actual drawing and camera fog', () => {
  const witnesses = [
    { at: 10, x: 114.49167762756348, z: -9.801540737152099, y: 0.8355403826597486, bilinear: 1.0507383408111823 },
    { at: 20, x: 46.045549392700195, z: -108.53816902160645, y: 1.1391468675184453, bilinear: 1.4997664889319497 },
  ];

  async function setup(witness: typeof witnesses[number]) {
    const f = fixture(witness.at), lib = cases();
    const snapshot = { status: f.data.status, front: f.front, frontCount: f.data.frontCount };
    const source: SurfaceSource = { grid: f.data.waterGrid, bedRevision: 0,
      get time() { return snapshot.status.seaTime; }, get revision() { return snapshot.status; },
      write: out => out.set(f.surface), writeBed: out => out.fill(-5) };
    const water = new WaterSurface(source); water.setLook('rich');
    const barrel = new SweptBarrel(water, async () => lib); barrel.setSpot('padang'); await barrel.ready;
    const host = { snapshot, heightAt: (x: number, z: number) => sampleSurfaceHeight(f.surface, f.data.waterGrid, x, z) } as unknown as SurfZoneHost;
    // Exercise the real PhysicalMode method and actual SweptBarrel draw, without starting or advancing a solver.
    const mode = Object.create(PhysicalMode.prototype) as PhysicalMode;
    const camera = new PerspectiveCamera(); camera.position.set(witness.x, witness.y, witness.z);
    const query = new RenderedCavityQuery();
    Object.assign(mode, { host, config: f.data.config, water, sweptBarrel: barrel, shown: true, swept: true,
      camera: { camera }, renderedCavity: query });
    return { f, water, barrel, host, snapshot, mode, camera, query, dispose: () => { barrel.dispose(); water.dispose(); } };
  }

  for (const witness of witnesses) it(`proves raw underwater→rendered cavity air on frozen state ${witness.at}, with contact oracle`, async () => {
    const t = await setup(witness);
    try {
      expect(t.host.heightAt(witness.x, witness.z)).toBeCloseTo(witness.bilinear, 12);
      expect(t.mode.cameraBelowSurface()).toBe(true); // No current drawing yet: exact old fallback.
      t.water.update(); t.mode.drawBarrel();
      const contact = new SweptContact(cases(), BARREL_SLOPE.padang!, { bucket: Infinity });
      contact.update(t.f.front, t.f.data.frontCount, t.f.data.config.tide, t.f.heightAt);
      const hit = createContactHit();
      expect(contact.query(witness.x, witness.y, witness.z, hit)).toBe(true);
      expect(hit.inWater).toBe(false); expect(hit.floorY).toBeLessThan(witness.y); expect(hit.ceilingY).toBeGreaterThan(witness.y);
      expect(t.mode.cameraBelowSurface()).toBe(false);
      const position = t.barrel.mesh.mesh.geometry.getAttribute('position') as BufferAttribute;
      const version = position.version;
      t.water.update(); t.mode.drawBarrel();
      expect(position.version).toBe(version); expect(t.mode.cameraBelowSurface()).toBe(false); // Real held draw.
      expect(t.mode.cameraBelowSurface(0.1, new Vector3(witness.x, hit.floorY - 0.2, witness.z))).toBe(true);
      expect(t.mode.cameraBelowSurface(0.1, new Vector3(witness.x, 20, witness.z))).toBe(false);
    } finally { t.dispose(); }
  });

  it('falls back before a new status or same-time restore is redrawn, and accepts the real view override', async () => {
    const witness = witnesses[0], t = await setup(witness);
    try {
      t.water.update(); t.mode.drawBarrel();
      const highView = new Vector3(witness.x, 20, witness.z), cavityView = t.camera.position.clone();
      t.camera.position.copy(highView);
      expect(t.mode.cameraBelowSurface()).toBe(false);
      expect(t.mode.cameraBelowSurface(0.1, cavityView)).toBe(false);
      for (const increment of [1 / 60, 0]) {
        t.snapshot.status = { ...t.snapshot.status, seaTime: t.snapshot.status.seaTime + increment };
        expect(t.mode.cameraBelowSurface(0.1, cavityView)).toBe(true);
        t.water.update();
        expect(t.mode.cameraBelowSurface(0.1, cavityView)).toBe(true);
        t.mode.drawBarrel();
        expect(t.mode.cameraBelowSurface(0.1, cavityView)).toBe(false);
      }
      const scan = vi.spyOn(t.query, 'contains');
      expect(t.mode.cameraBelowSurface(0.1, highView)).toBe(false); expect(scan).not.toHaveBeenCalled();
    } finally { t.dispose(); }
  });

  it('rejects hidden/empty/unuploaded/maskless drawing without changing the raw fallback', async () => {
    const witness = witnesses[0], t = await setup(witness);
    try {
      t.water.update(); t.mode.drawBarrel(); expect(t.mode.cameraBelowSurface()).toBe(false);
      const mesh = t.barrel.mesh.mesh, geometry = mesh.geometry, count = geometry.drawRange.count;
      mesh.visible = false; expect(t.mode.cameraBelowSurface()).toBe(true); mesh.visible = true;
      geometry.setDrawRange(1, count); expect(t.mode.cameraBelowSurface()).toBe(true); geometry.setDrawRange(0, count);
      geometry.setDrawRange(0, 0); expect(t.mode.cameraBelowSurface()).toBe(true); geometry.setDrawRange(0, count);
      const mask = (t.water.materialUniforms.waterBarrelMask.value as DataTexture).image.data as Uint8Array;
      const grid = t.water.barrelMaskGrid, node = Math.floor((witness.z - grid.zMin) / grid.spacing) * grid.nx
        + Math.floor((witness.x - grid.xMin) / grid.spacing);
      const previous = mask[node]; mask[node] = 0;
      expect(t.water.barrelMaskActive).toBe(true); expect(t.mode.cameraBelowSurface()).toBe(true);
      mask[node] = 254; expect(t.mode.cameraBelowSurface()).toBe(true);
      mask[node] = previous; expect(t.mode.cameraBelowSurface()).toBe(false);
      (geometry.getAttribute('position') as BufferAttribute).needsUpdate = true;
      expect(t.mode.cameraBelowSurface()).toBe(true);
      t.water.setBarrelMask(null); expect(t.mode.cameraBelowSurface()).toBe(true);
      t.snapshot.front = new Float32Array(); t.snapshot.frontCount = 0;
      t.snapshot.status = { ...t.snapshot.status };
      t.water.update(); t.mode.drawBarrel(); expect(t.mode.cameraBelowSurface()).toBe(true);
    } finally { t.dispose(); }
  });
});
