/** Fixed before any run; these are QA eligibility choices, not a promise that source continuity survives. */
export const MATURE_CRITERIA=Object.freeze({minimumJoinedSpanMeters:12,minimumJoinedEndDistanceMeters:4.5,
  minimumFrontEndpointDistanceMeters:3,minimumPhysicalEdgeDistanceMeters:3,minimumLocalWeight:.9,
  minimumLocalFormed:.9,minimumLifeFraction:.2,maximumLifeFraction:.75,minimumAirGapMeters:.15});

/** Same exact source control, measured against the matching actual drawn front. No tessellation or source mutation. */
export function sourceMaturityAt(front,count,loft,grid,k){
  if(!loft||!Number.isInteger(k)||k<0||k>=count)return null;
  const o=9*k,id=front[o+2],x=front[o],sigma=front[o+3];
  if(!Number.isFinite(id)||!Number.isFinite(x)||!Number.isFinite(sigma))return null;
  let first=k,last=k;
  while(first>0&&front[9*(first-1)+2]===id)first--;
  while(last+1<count&&front[9*(last+1)+2]===id)last++;
  // Fragmented blocks are invalid authority, not independent copies of one selectable front.
  for(let i=0;i<count;i++)if((i<first||i>last)&&front[9*i+2]===id)return null;
  let strip=-1;
  for(let s=0;s+1<loft.sliceCount;s++)if(loft.sliceJoined[s]===1&&loft.sliceFront[s]===id&&loft.sliceFront[s+1]===id&&loft.sliceSigma[s]<=sigma&&sigma<loft.sliceSigma[s+1]){
    if(strip!==-1)return null;strip=s;
  }
  if(strip<0)return null;
  let runFirst=strip,runLast=strip+1;
  while(runFirst>0&&loft.sliceJoined[runFirst-1]===1&&loft.sliceFront[runFirst-1]===id)runFirst--;
  while(runLast+1<loft.sliceCount&&loft.sliceJoined[runLast]===1&&loft.sliceFront[runLast+1]===id)runLast++;
  const values=keys=>keys.map(key=>[loft[key][strip],loft[key][strip+1]]);
  const [phase,overturned,weight,formed,life]=values(['slicePhase','sliceOverturned','sliceWeight','sliceFormed','sliceLife']);
  const xMax=grid.xMin+(grid.nx-1)*grid.spacing;
  return {front:id,sourceX:x,sourceSigma:sigma,sourceRecordIndex:k,strip,runFirst,runLast,
    joinedSpanMeters:loft.sliceSigma[runLast]-loft.sliceSigma[runFirst],
    joinedEndDistanceMeters:Math.min(sigma-loft.sliceSigma[runFirst],loft.sliceSigma[runLast]-sigma),
    frontEndpointDistanceMeters:Math.min(sigma-front[9*first+3],front[9*last+3]-sigma),
    physicalEdgeDistanceMeters:Math.min(x-grid.xMin,xMax-x),phase,overturned,weight,formed,life};
}

export function meetsMatureCriteria(m,c){
  if(!m||!c||![m.phase,m.overturned,m.weight,m.formed,m.life].every(v=>Array.isArray(v)&&v.length===2))return false;
  const atLeast=(v,min)=>Number.isFinite(v)&&v>=min;
  return atLeast(m.joinedSpanMeters,c.minimumJoinedSpanMeters)&&atLeast(m.joinedEndDistanceMeters,c.minimumJoinedEndDistanceMeters)
    &&atLeast(m.frontEndpointDistanceMeters,c.minimumFrontEndpointDistanceMeters)&&atLeast(m.physicalEdgeDistanceMeters,c.minimumPhysicalEdgeDistanceMeters)
    &&m.phase.every(v=>v===1)&&m.overturned.every(v=>v===1)&&m.weight.every(v=>atLeast(v,c.minimumLocalWeight))
    &&m.formed.every(v=>atLeast(v,c.minimumLocalFormed))&&m.life.every(v=>atLeast(v,c.minimumLifeFraction)&&v<=c.maximumLifeFraction);
}

/** Source record order is the sole priority. A later prettier witness can never displace the first eligible one. */
export function firstMatureSource(count,measure,qualifies,witness){
  for(let index=0;index<count;index++){const metrics=measure(index);if(!qualifies(metrics))continue;const view=witness(metrics);if(view)return{index,metrics,witness:view};}
  return null;
}

/** Camera planning can only read the controls transported with the exact actual drawn revision. */
export function drawnControlAt(authority,revision,front,sourceX){
  if(!authority||authority.revision!==revision)return null;
  for(let k=0;k<authority.count;k++){
    const o=9*k;
    if(authority.front[o+2]===front&&authority.front[o]===sourceX)return{x:sourceX,z:authority.front[o+1],sigma:authority.front[o+3],tau:authority.front[o+4]};
  }
  return null;
}

/** Exact typed bytes, including NaN sentinel bits; no decimal-number arrays cross CDP for checkpoint fields. */
export function encodeTypedPrefix(view,count){
  const types=['Float32Array','Float64Array','Uint8Array','Uint16Array','Uint32Array','Int32Array'];
  if(!ArrayBuffer.isView(view)||!types.includes(view.constructor.name)||!Number.isInteger(count)||count<0||count>view.length)throw Error('Invalid checkpoint typed prefix');
  if(new Uint8Array(new Uint16Array([1]).buffer)[0]!==1)throw Error('Checkpoint requires explicit little-endian authority');
  const bytes=new Uint8Array(view.buffer,view.byteOffset,count*view.BYTES_PER_ELEMENT),parts=[];
  for(let k=0;k<bytes.length;k+=16384)parts.push(String.fromCharCode(...bytes.subarray(k,k+16384)));
  return{type:view.constructor.name,length:count,byteLength:bytes.length,endian:'little',encoding:'base64',base64:btoa(parts.join(''))};
}

/** QA state only: the first eligible front and source column never change after selection. */
export class MatureFixedFrontLifecycle {
  front = null;
  sourceX = null;
  events = [];
  quietRows = 0;
  opened = false;
  touchdown = false;
  retired = false;
  lock(candidate, seaTime) {
    if (this.front !== null || candidate?.eligible !== true || !candidate?.verifiedAir || !Number.isFinite(candidate.front) || !Number.isFinite(candidate.sourceX)) return false;
    this.front = candidate.front; this.sourceX = candidate.sourceX;
    this.events.push({ kind: 'selected', seaTime, front: this.front, sourceX: this.sourceX });
    return true;
  }
  observe(row) {
    if (this.front === null) return [];
    if (row.front !== this.front || row.sourceX !== this.sourceX) throw Error('Tracked source identity changed');
    const events = [], add = kind => { const e = {kind,seaTime:row.seaTime,front:this.front,sourceX:this.sourceX};events.push(e);this.events.push(e); };
    if (!this.opened && row.phase === 1 && row.verifiedAir && row.indexedLift) { this.opened=true;add('first-open'); }
    if (this.opened && !this.touchdown && row.phase === 2) { this.touchdown=true;add('first-touchdown'); }
    // An overlap cut during an open phase is not retirement. After actual same-column touchdown, removal
    // must persist for three fresh snapshots; distinguish rendered-column removal from source/front loss.
    const gone = this.touchdown && (!row.sourcePresent || (!row.indexedLift && row.phase !== 1));
    this.quietRows = gone ? this.quietRows+1 : 0;
    if (!this.retired && this.quietRows>=3) { this.retired=true;add('first-retired'); }
    return events;
  }
  summary() { return {front:this.front,sourceX:this.sourceX,opened:this.opened,touchdown:this.touchdown,retired:this.retired,complete:this.opened&&this.touchdown&&this.retired,events:this.events}; }
}

  const criteria=options.matureCriteria;

    const c=row.coverage;if(options.candidate&&c.supported===1&&c.active===1&&(!c.waterMark||c.sweptMark!==(c.barrelVisible&&v.drawRange.count>0)||!c.repairVisible))row.candidateFailures.push('Active fallback state mismatch');
    const predicate=state.lastPredicate;
    if(!predicate||predicate.seaTime!==status.seaTime||!predicate.suppliedPositionMatchesCamera||!row.cavity.predicateMatchesCamera||!predicate.supplied.every((value,k)=>value===eye[k]))row.qaInvalid.push('Classifier camera/source provenance differs from actual rendered camera');
    if(phase===1&&observed.indexedLift&&(!verifiedAir||!targetAir||row.view.projectedOpenLiftedVertices===0))row.qaInvalid.push('Tracked open-column camera has no fresh verified eye/target air or projected opening');
    if(options.fogCandidate&&verifiedAir&&row.cavity.rawUnderwater&&!fogDrawFresh)row.qaInvalid.push('Fresh fog draw provenance unavailable for verified air');
    const fogEligible=row.qaInvalid.length===0;
    if(fogEligible&&options.fogCandidate&&verifiedAir&&row.cavity.rawUnderwater&&predicate.result!==false)row.candidateFailures.push('Combined opaque indexed air retained underwater classifier');
    if(fogEligible&&Boolean(fog)!==predicate.result)row.candidateFailures.push('Actual scene fog differs from original classifier');
    row.failures=[...row.candidateFailures,...row.qaInvalid];row.pass=row.failures.length===0;row.inspectMs=performance.now()-began;state.rows.push(row);
    for(const e of events)checkpoint(row,e.kind);
  }
  // This hook reads only at the real default-scene clear, AFTER original draw and fog classification.
  // It covers both legacy pre-draw classification and the separate combined post-draw fog candidate.
  const gl=d.canvas.getContext('webgl2'),clear=gl.clear.bind(gl);
  gl.clear=bits=>{if(state.active&&(bits&gl.COLOR_BUFFER_BIT)&&gl.getParameter(gl.DRAW_FRAMEBUFFER_BINDING)===null)recordRenderedFrame();return clear(bits);};
  const metadata=()=>({cached:state.lastDraw?.cached??false,seaTime:mode.host.snapshot.status.seaTime,versions:versions(),coverage:coverage(),drawn:mode.sweptBarrel?.drawn?.revision===mode.host.snapshot.status});
  function actualControls() {
    if(!lab.clock.paused||mode.host.outstandingSteps)throw Error('Controls require settled paused host');
    state.finalControls=true;
    const camera=mode.camera.camera,clock=mode.host.snapshot.status.seaTime;
    const render=name=>{d.renderView(camera);const result={name,...metadata()};if(result.seaTime!==clock)throw Error('Control advanced clock');state.controls.push(result);return result;};
    render('active-settled');const a=render('active-cached-1'),b=render('active-cached-2');if(!a.cached||!b.cached||JSON.stringify(a.versions)!==JSON.stringify(b.versions))throw Error('Held real cached draws rebuilt/uploaded geometry');
    if(options.candidate){water.setBarrelScreenFallback(false);const u=render('unsupported-real-cache');if(u.coverage.waterMark||u.coverage.sweptMark||u.coverage.repairVisible)throw Error('Unsupported real cache did not clear');if(JSON.stringify(u.versions)!==JSON.stringify(b.versions))throw Error('Unsupported toggle uploaded geometry');water.setBarrelScreenFallback(true);const r=render('reactivated-real-cache');if(JSON.stringify(r.versions)!==JSON.stringify(b.versions))throw Error('Reactivation uploaded geometry');if(r.coverage.active===1&&(!r.coverage.waterMark||!r.coverage.repairVisible))throw Error('Reactivation missing fallback');}
    mode.setVisible(false);const hidden=render('mode-hidden');if(hidden.coverage.active!==0||hidden.coverage.barrelVisible||hidden.coverage.repairVisible||hidden.coverage.waterMark||hidden.coverage.sweptMark)throw Error('Hidden mode did not clear fallback');mode.setVisible(true);render('mode-visible-restored');
    mode.sweptBarrel.setSpot(undefined);const off=render('spot-off');if(off.coverage.active!==0||off.coverage.barrelVisible||off.coverage.repairVisible||off.coverage.waterMark||off.coverage.sweptMark)throw Error('Spot transition did not clear fallback');mode.sweptBarrel.setSpot('padang',true);render('spot-restored');
    return state.controls;
  }
  const drain=()=>({rows:state.rows.splice(0),published:window.__tubePublished.splice(0),events:state.events.splice(0),checkpoints:state.checkpoints.splice(0)});
  window.__tubeMovingMature={start:()=>{state.active=true;state.started=performance.now();window.__tubeCaptureActive=true;},drain,controls:actualControls,
    stop:()=>{state.active=false;window.__tubeCaptureActive=false;state.stopped=performance.now();return{...drain(),started:state.started,stopped:state.stopped,frames:state.frames,cachedDraws:state.cached,lifecycle:tracker.summary(),classifierCalls:state.classifierCalls,controls:state.controls,selection:state.selection,lastSelectionScan:state.lastSelectionScan};}};
}
