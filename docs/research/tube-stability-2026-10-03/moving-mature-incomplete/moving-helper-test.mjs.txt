 const first=firstMatureSource(list.length,k=>list[k],m=>meetsMatureCriteria(m,MATURE_CRITERIA),m=>({gap:m.sourceX}));assert.equal(first.index,1);
 const skipped=firstMatureSource(list.length,k=>list[k],m=>meetsMatureCriteria(m,MATURE_CRITERIA),m=>m===valid?null:{gap:.5});assert.equal(skipped.index,2);
 assert.equal(firstMatureSource(list.length,k=>list[k],()=>false,()=>({})),null);
});
test('exact typed active-prefix transport preserves F32 bits and NaN sentinels without unused tails',()=>{
 const all=new Float32Array([99,1.1,NaN,-0,8]),view=all.subarray(1,4),packed=encodeTypedPrefix(view,3);assert.equal(packed.type,'Float32Array');assert.equal(packed.length,3);assert.equal(packed.byteLength,12);assert.equal(packed.endian,'little');assert.deepEqual(Buffer.from(packed.base64,'base64'),Buffer.from(view.buffer,view.byteOffset,12));assert.throws(()=>encodeTypedPrefix(view,4),/Invalid/);assert.throws(()=>encodeTypedPrefix(new DataView(all.buffer),1),/Invalid/);
 const indices=new Uint32Array([1,2,3,999]);assert.equal(encodeTypedPrefix(indices,3).byteLength,12);
});

test('tracked camera uses prior-draw control sigma rather than current host; both buffers remain unchanged',()=>{
 const f=fixture(),revision={seaTime:170},current=Float32Array.from(f.front);
 for(let k=0;k<f.count;k++)current[9*k+3]+=2;
 const beforeDrawn=Buffer.from(f.front.buffer).toString('hex'),beforeCurrent=Buffer.from(current.buffer).toString('hex');
 const authority={revision,count:f.count,front:f.front},control=drawnControlAt(authority,revision,24,8);
 assert.equal(control.sigma,8);assert.equal(current[9*8+3],10);
 assert.equal(sourceMaturityAt(f.front,f.count,f.loft,f.grid,8).strip,16);
 assert.equal(sourceMaturityAt(current,f.count,f.loft,f.grid,8).strip,20);
 assert.equal(drawnControlAt(authority,{seaTime:170},24,8),null);
 assert.equal(drawnControlAt(authority,revision,24,99),null);

});
