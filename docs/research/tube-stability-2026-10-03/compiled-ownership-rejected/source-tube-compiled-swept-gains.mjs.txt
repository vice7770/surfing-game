import{readFileSync,writeFileSync}from'node:fs';
import{BufferGeometry,BufferAttribute,Mesh,MeshBasicMaterial,DoubleSide,PerspectiveCamera,Raycaster,Vector2,Vector3,Triangle}from'/Users/regina/Desktop/Projects/surfing-game/node_modules/three/build/three.module.js';
const root='/private/tmp',data=JSON.parse(readFileSync(root+'/tube-compiled-swept-gains.json'));
const count=(o,k)=>o[k]=(o[k]??0)+1;
function alpha(M,g,x,z){const gx=Math.min(g.nx-1,Math.max(0,(x-g.xMin)/g.spacing)),gz=Math.min(g.nz-1,Math.max(0,(z-g.zMin)/g.spacing)),i=Math.min(g.nx-2,Math.floor(gx)),j=Math.min(g.nz-2,Math.floor(gz)),tx=gx-i,tz=gz-j,n=j*g.nx+i;return((M[n]*(1-tx)+M[n+1]*tx)*(1-tz)+(M[n+g.nx]*(1-tx)+M[n+g.nx+1]*tx)*tz)/255;}
function layer(j){return j<3?'backExtension':j<35?'back':j<=67?'roofTop':j<=91?'roofUnder':j<=115?'face':j<131?'frontRest':'frontExtension';}
const report={schema:1,method:'Exact full gained swept-ID counts. At most512 deterministic screen samples per camera, raycast actual indexed Float32 world-position triangles using exact recorded perspective camera. Swept vertex projection is identity vec3(position). Filter nearest CPU ray hits using node-centred bilinear mask and supplied GPU dither +/-1/255 uncertainty; classify sampled front/profile rows, lift/weights and depth relative to retained GPU-displaced original water. This is bounded attribution, not exact GPU triangle-ID instrumentation.',pairs:[]};
for(const rec of data.samples){const G=JSON.parse(readFileSync(`${root}/tube-geometry-final/at-${rec.at}.current.json`)),L=G.loft,geometry=new BufferGeometry();geometry.setAttribute('position',new BufferAttribute(Float32Array.from(L.positions),3));geometry.setIndex(new BufferAttribute(Uint32Array.from(L.indices),1));geometry.computeBoundingSphere();const mesh=new Mesh(geometry,new MeshBasicMaterial({side:DoubleSide}));mesh.updateMatrixWorld(true);const camera=new PerspectiveCamera();camera.position.fromArray(rec.camera.position);camera.quaternion.fromArray(rec.camera.quaternion);camera.projectionMatrix.fromArray(rec.camera.projection);camera.projectionMatrixInverse.copy(camera.projectionMatrix).invert();camera.updateMatrixWorld(true);const caster=new Raycaster(),a=new Vector3(),b=new Vector3(),c=new Vector3(),bary=new Vector3(),summary={};const samples=[];
for(const probe of rec.samples){const n=probe.index,x=n%rec.width,y=Math.floor(n/rec.width),dither=probe.gpuDitherQuantized??((52.9829189*(((x+.5)*.06711056+(y+.5)*.00583715)%1))%1);caster.setFromCamera(new Vector2(2*(x+.5)/rec.width-1,2*(y+.5)/rec.height-1),camera);const hits=caster.intersectObject(mesh,false),selected=hits.find(h=>alpha(G.mask,G.maskGrid,h.point.x,h.point.z)>dither);if(!selected){count(summary,'noSurvivingCenterHit');continue;}const t=selected.faceIndex*3,vs=L.indices.slice(t,t+3),s=Math.min(...vs.map(v=>Math.floor(v/134))),j=Math.min(...vs.map(v=>v%134));a.fromArray(L.positions,3*vs[0]);b.fromArray(L.positions,3*vs[1]);c.fromArray(L.positions,3*vs[2]);Triangle.getBarycoord(selected.point,a,b,c,bary);const w=bary.toArray(),mix=field=>vs.reduce((sum,vertex,k)=>sum+w[k]*L[field][vertex],0),al=alpha(G.mask,G.maskGrid,selected.point.x,selected.point.z),waterDistance=probe.waterWorld?caster.ray.origin.distanceTo(new Vector3().fromArray(probe.waterWorld)):null,behind=waterDistance===null?null:selected.distance-waterDistance;count(summary,'accepted');count(summary,'layer_'+layer(j));count(summary,'front_'+L.sliceFront[s]);count(summary,mix('lift')<.01?'liftUnder1percent':mix('lift')>.5?'liftOverHalf':'partialLift');count(summary,al-dither>1/255?'maskSelectionMarginOverQuantization':'maskSelectionNearQuantization');if(behind!==null)count(summary,behind>.02?'fartherThanOriginalWater2cm':behind<-.02?'nearerThanOriginalWater2cm':'within2cmOriginalWater');samples.push({...probe,swept:{point:selected.point.toArray(),distance:selected.distance,strip:s,row:j,layer:layer(j),front:L.sliceFront[s],sigma:L.sliceSigma[s],phase:L.slicePhase[s],life:L.sliceLife[s],formed:L.sliceFormed[s],weight:L.sliceWeight[s],collapse:L.sliceCollapse[s],mouth:L.sliceMouth[s],lift:mix('lift'),sheetWeight:mix('sheetWeight'),maskAlpha:al,maskDitherMargin:al-dither,bary:w},fartherThanOriginalWaterMeters:behind});}
report.pairs.push({at:rec.at,view:rec.view,totalNewSwept:rec.totalNewSwept,from:rec.from,sampled:rec.samples.length,summary,probes:samples.slice(0,12),sampling:'Deterministic equal quantiles across gained pure swept pixels (bottom-left framebuffer ordering), at most512 per camera.',sources:rec.sources});console.log(rec.at,rec.view,rec.totalNewSwept,summary);geometry.dispose();mesh.material.dispose();}
writeFileSync(root+'/tube-compiled-swept-attribution.json',JSON.stringify(report,null,2)+'\n');
