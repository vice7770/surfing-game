function heightAt(data,grid,x,z){const gx=(x-grid.xMin)/grid.spacing,gz=(z-grid.zMin)/grid.spacing,i0=Math.floor(gx),j0=Math.floor(gz),wx=weights(gx-i0),wz=weights(gz-j0);let h=0;for(let j=0;j<4;j++)for(let i=0;i<4;i++){const row=Math.min(grid.nz-1,Math.max(0,j0+j-1)),col=Math.min(grid.nx-1,Math.max(0,i0+i-1));h+=wz[j]*wx[i]*data[(row*grid.nx+col)*2];}return h;}
